```python
def apply_phase6a_schema_migrations(engine: Engine) -> None:
    """
    Idempotent schema migration helper invoked on FastAPI startup after
    `Base.metadata.create_all(bind=engine)`. Safely adds Phase 6A columns
    to existing `findings` and `scans` tables without dropping any data.
    """
    Base.metadata.create_all(bind=engine)

    inspector = inspect(engine)
    existing_tables = set(inspector.get_table_names())

    findings_migrations: list[tuple[str, str]] = [
        ("cve_id", "VARCHAR(64)"),
        ("matched_engines_json", "TEXT DEFAULT '[]' NOT NULL"),
        ("engine_count", "INTEGER DEFAULT 1 NOT NULL"),
        ("confidence_level", "VARCHAR(32) DEFAULT 'medium' NOT NULL"),
        ("epss_score", "FLOAT"),
        ("epss_percentile", "FLOAT"),
        ("is_kev", "BOOLEAN DEFAULT FALSE NOT NULL"),
        ("kev_due_date", "VARCHAR(32)"),
        ("effective_priority_score", "FLOAT DEFAULT 0.0 NOT NULL"),
    ]

    scans_migrations: list[tuple[str, str]] = [
        ("raw_findings_count", "INTEGER DEFAULT 0 NOT NULL"),
        ("correlated_findings_count", "INTEGER DEFAULT 0 NOT NULL"),
        ("kev_findings_count", "INTEGER DEFAULT 0 NOT NULL"),
    ]

    with engine.begin() as conn:
        if "findings" in existing_tables:
            existing_finding_cols = {col["name"] for col in inspector.get_columns("findings")}
            for col_name, col_ddl in findings_migrations:
                if col_name not in existing_finding_cols:
                    conn.execute(text(f"ALTER TABLE findings ADD COLUMN {col_name} {col_ddl}"))

            conn.execute(
                text("CREATE INDEX IF NOT EXISTS ix_findings_cve_id ON findings (cve_id)")
            )
            conn.execute(
                text(
                    "CREATE INDEX IF NOT EXISTS ix_findings_effective_priority_score "
                    "ON findings (effective_priority_score)"
                )
            )
            conn.execute(
                text("CREATE INDEX IF NOT EXISTS ix_findings_is_kev ON findings (is_kev)")
            )

        if "scans" in existing_tables:
            existing_scan_cols = {col["name"] for col in inspector.get_columns("scans")}
            for col_name, col_ddl in scans_migrations:
                if col_name not in existing_scan_cols:
                    conn.execute(text(f"ALTER TABLE scans ADD COLUMN {col_name} {col_ddl}"))
```

```python
from __future__ import annotations

import hashlib
import json
import re
from dataclasses import dataclass, field
from typing import Any, Iterable

CVE_REGEX = re.compile(r"\b(CVE-\d{4}-\d{4,7})\b", re.IGNORECASE)
GHSA_REGEX = re.compile(r"\b(GHSA-[a-z0-9]{4}-[a-z0-9]{4}-[a-z0-9]{4})\b", re.IGNORECASE)
CWE_REGEX = re.compile(r"\bCWE-(\d{1,5})\b", re.IGNORECASE)

SCA_ENGINES = {"osv-scanner", "osv_scanner", "pip-audit", "pip_audit", "trivy", "trivy_sca"}
SECRET_ENGINES = {"gitleaks", "trufflehog", "detect-secrets", "detect_secrets"}
SAST_IAC_ENGINES = {
    "semgrep",
    "bandit",
    "njsscan",
    "shellcheck",
    "checkov",
    "hadolint",
    "openrouter_review",
}

SEVERITY_RANK: dict[str, int] = {
    "critical": 5,
    "high": 4,
    "medium": 3,
    "low": 2,
    "info": 1,
    "informational": 1,
}

CONFIDENCE_RANK: dict[str, int] = {
    "high": 3,
    "medium": 2,
    "low": 1,
}


@dataclass
class CorrelationSummary:
    raw_findings_count: int
    correlated_findings_count: int
    deduplicated_count: int
    multi_engine_confirmed_count: int
    findings: list[dict[str, Any]] = field(default_factory=list)


def normalize_rel_path(file_path: str | None) -> str:
    if not file_path:
        return "unknown"
    cleaned = file_path.strip().replace("\\", "/")
    while cleaned.startswith("./"):
        cleaned = cleaned[2:]
    cleaned = cleaned.lstrip("/")
    return cleaned or "unknown"


def normalize_engine_name(engine: str | None) -> str:
    if not engine:
        return "unknown"
    e = engine.strip().lower().replace("_", "-")
    if e == "openrouter-review":
        return "openrouter_review"
    if e == "trivy-license":
        return "trivy_license"
    return e


def normalize_severity(severity: str | None) -> str:
    if not severity:
        return "medium"
    s = severity.strip().lower()
    if s in ("critical", "crit"):
        return "critical"
    if s in ("high", "error"):
        return "high"
    if s in ("medium", "moderate", "warning", "warn"):
        return "medium"
    if s in ("low",):
        return "low"
    if s in ("info", "informational", "note"):
        return "info"
    return "medium"


def normalize_confidence(confidence: str | None) -> str:
    if not confidence:
        return "medium"
    c = confidence.strip().lower()
    if c in ("high", "confirmed", "verified"):
        return "high"
    if c in ("low",):
        return "low"
    return "medium"


def extract_primary_cve(raw: dict[str, Any]) -> str | None:
    """
    Inspect explicit fields (`cve_id`, `cve`, `rule_id`, `aliases`, `title`, `description`)
    and return the canonical uppercase `CVE-YYYY-NNNNN` if present.
    """
    candidates: list[str] = []

    for key in ("cve_id", "cve", "advisory_id", "vulnerability_id", "rule_id", "id"):
        val = raw.get(key)
        if isinstance(val, str):
            candidates.append(val)

    aliases = raw.get("aliases")
    if isinstance(aliases, (list, tuple, set)):
        for item in aliases:
            if isinstance(item, str):
                candidates.append(item)
    elif isinstance(aliases, str):
        candidates.append(aliases)

    for key in ("title", "description", "message"):
        val = raw.get(key)
        if isinstance(val, str):
            candidates.append(val)

    for text_item in candidates:
        match = CVE_REGEX.search(text_item)
        if match:
            return match.group(1).upper()

    return None


def extract_primary_advisory_id(raw: dict[str, Any]) -> str:
    cve = extract_primary_cve(raw)
    if cve:
        return cve

    candidates: list[str] = []
    for key in ("advisory_id", "vulnerability_id", "rule_id", "id", "title"):
        val = raw.get(key)
        if isinstance(val, str):
            candidates.append(val)

    aliases = raw.get("aliases")
    if isinstance(aliases, (list, tuple, set)):
        candidates.extend(str(a) for a in aliases if a)

    for text_item in candidates:
        ghsa_match = GHSA_REGEX.search(text_item)
        if ghsa_match:
            return ghsa_match.group(1).upper()

    for key in ("rule_id", "advisory_id", "id", "title"):
        val = raw.get(key)
        if isinstance(val, str) and val.strip():
            return val.strip().upper()

    return "UNKNOWN-ADVISORY"


def extract_canonical_cwe(raw: dict[str, Any]) -> str | None:
    for key in ("cwe", "cwe_id", "rule_id", "title", "description"):
        val = raw.get(key)
        if isinstance(val, str):
            match = CWE_REGEX.search(val)
            if match:
                return f"CWE-{int(match.group(1))}"
        elif isinstance(val, list):
            for item in val:
                if isinstance(item, str):
                    match = CWE_REGEX.search(item)
                    if match:
                        return f"CWE-{int(match.group(1))}"
    return None


def infer_sca_package_coordinates(raw: dict[str, Any]) -> tuple[str, str, str]:
    """
    Returns `(ecosystem, package_name_lower, installed_version)` from structured
    fields or falls back to parsing title/description/file_path patterns.
    """
    ecosystem = str(raw.get("ecosystem") or "").strip().lower()
    pkg_name = str(
        raw.get("package_name") or raw.get("package") or raw.get("pkg_name") or ""
    ).strip().lower()
    installed_version = str(
        raw.get("installed_version") or raw.get("version") or raw.get("pkg_version") or ""
    ).strip().lower()

    file_path = normalize_rel_path(raw.get("file_path") or raw.get("path"))
    if not ecosystem:
        if file_path.endswith(("requirements.txt", "pipfile.lock", "poetry.lock", "pyproject.toml")):
            ecosystem = "pypi"
        elif file_path.endswith(("package.json", "package-lock.json", "yarn.lock", "pnpm-lock.yaml")):
            ecosystem = "npm"
        elif file_path.endswith(("go.mod", "go.sum")):
            ecosystem = "go"
        elif file_path.endswith(("cargo.toml", "cargo.lock")):
            ecosystem = "cargo"
        else:
            ecosystem = "generic"

    if not pkg_name or not installed_version:
        text_blob = f"{raw.get('title', '')} {raw.get('description', '')}"
        at_match = re.search(r"\b([a-zA-Z0-9_.\-]+)@([0-9][a-zA-Z0-9_.\-+]*)\b", text_blob)
        if at_match:
            if not pkg_name:
                pkg_name = at_match.group(1).lower()
            if not installed_version:
                installed_version = at_match.group(2).lower()
        else:
            in_match = re.search(
                r"\b([a-zA-Z0-9_.\-]+)\s+\(?([0-9]+\.[0-9]+(?:\.[0-9a-zA-Z.\-]+)?)\)?",
                text_blob,
            )
            if in_match:
                if not pkg_name:
                    pkg_name = in_match.group(1).lower()
                if not installed_version:
                    installed_version = in_match.group(2).lower()

    return (
        ecosystem or "generic",
        pkg_name or "unknown-pkg",
        installed_version or "unknown-ver",
    )


def infer_secret_family(raw: dict[str, Any]) -> str:
    text_blob = " ".join(
        str(raw.get(k) or "") for k in ("secret_type", "rule_id", "title", "description")
    ).lower()
    if "aws" in text_blob:
        return "aws-credential"
    if "github" in text_blob or "ghp_" in text_blob:
        return "github-token"
    if "slack" in text_blob:
        return "slack-token"
    if "stripe" in text_blob:
        return "stripe-key"
    if "private" in text_blob and "key" in text_blob:
        return "private-key"
    if "jwt" in text_blob:
        return "jwt-secret"
    if "postgres" in text_blob or "mysql" in text_blob or "mongodb" in text_blob or "database" in text_blob:
        return "database-uri"
    return "generic-secret"


def compute_raw_fingerprint(raw: dict[str, Any]) -> str:
    """
    Preserves existing raw fingerprint if already generated by the engine runner,
    or computes a deterministic SHA-256 fingerprint.
    """
    existing = raw.get("fingerprint")
    if isinstance(existing, str) and existing.strip():
        return existing.strip()

    engine = normalize_engine_name(raw.get("engine"))
    file_path = normalize_rel_path(raw.get("file_path") or raw.get("path"))
    rule_id = str(raw.get("rule_id") or raw.get("cve_id") or raw.get("title") or "rule").strip()
    line = int(raw.get("line_number") or raw.get("line") or raw.get("start_line") or 0)
    raw_str = f"{engine}|{file_path}|{rule_id}|{line}"
    return hashlib.sha256(raw_str.encode("utf-8")).hexdigest()[:32]


def compute_canonical_fingerprint(
    category: str,
    file_path: str,
    canonical_rule_or_cve: str,
    line_or_pkg: str,
) -> str:
    payload = f"{category.lower()}|{normalize_rel_path(file_path)}|{canonical_rule_or_cve.upper()}|{line_or_pkg.lower()}"
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()[:32]


def _classify_category(raw: dict[str, Any], engine: str) -> str:
    cat = str(raw.get("category") or "").strip().lower()
    if cat in ("sca", "dependency", "dependencies", "vuln"):
        return "sca"
    if cat in ("secret", "secrets"):
        return "secret"
    if cat in ("license", "sbom"):
        return "license"
    if cat in ("iac", "container", "dockerfile"):
        return "iac"
    if cat in ("sast", "code", "ai_review"):
        return "sast"

    if engine in SCA_ENGINES:
        return "sca"
    if engine in SECRET_ENGINES:
        return "secret"
    if engine == "trivy_license":
        return "license"
    if engine in ("checkov", "hadolint"):
        return "iac"
    return "sast"


def build_correlation_key(raw: dict[str, Any]) -> tuple[str, ...]:
    engine = normalize_engine_name(raw.get("engine"))
    category = _classify_category(raw, engine)
    file_path = normalize_rel_path(raw.get("file_path") or raw.get("path"))
    line_number = int(raw.get("line_number") or raw.get("line") or raw.get("start_line") or 0)

    if category == "sca":
        ecosystem, pkg_name, installed_ver = infer_sca_package_coordinates(raw)
        advisory_id = extract_primary_advisory_id(raw)
        return ("sca", ecosystem, pkg_name, installed_ver, advisory_id)

    if category == "secret":
        snippet = str(raw.get("code_snippet") or raw.get("snippet") or raw.get("match") or "").strip()
        if snippet and line_number > 0:
            snippet_hash = hashlib.sha256(snippet.encode("utf-8")).hexdigest()[:16]
            return ("secret_snippet", file_path, str(line_number), snippet_hash)
        secret_family = infer_secret_family(raw)
        return ("secret_loc", file_path, str(line_number), secret_family)

    if category == "license":
        ecosystem, pkg_name, installed_ver = infer_sca_package_coordinates(raw)
        rule_id = str(raw.get("rule_id") or raw.get("title") or "license").strip().lower()
        return ("license", ecosystem, pkg_name, installed_ver, rule_id)

    # SAST & IaC correlation: bucket by (file_path, line // 3, canonical_cwe_id) when CWE is known
    cwe_id = extract_canonical_cwe(raw)
    if cwe_id and line_number > 0:
        line_bucket = line_number // 3
        return ("sast_cwe", file_path, str(line_bucket), cwe_id)

    rule_id = str(raw.get("rule_id") or raw.get("title") or "generic-rule").strip().lower()
    return ("sast_rule", engine, file_path, str(line_number), rule_id)


def _merge_finding_group(
    corr_key: tuple[str, ...],
    group: list[dict[str, Any]],
    suppressed_fingerprints: set[str],
) -> dict[str, Any]:
    """
    Merges 1..N raw findings sharing `corr_key` into a single canonical finding dict.
    """
    # Sort group so highest severity and longest description come first for base selection
    sorted_group = sorted(
        group,
        key=lambda item: (
            SEVERITY_RANK.get(normalize_severity(item.get("severity")), 0),
            len(str(item.get("description") or "")),
        ),
        reverse=True,
    )
    primary = dict(sorted_group[0])

    matched_engines = sorted(
        {normalize_engine_name(item.get("engine")) for item in group if item.get("engine")}
    )
    if not matched_engines:
        matched_engines = ["unknown"]
    engine_count = len(matched_engines)

    # Highest severity across all engines
    best_severity = max(
        (normalize_severity(item.get("severity")) for item in group),
        key=lambda sev: SEVERITY_RANK.get(sev, 0),
    )

    # Longest, most informative description
    best_description = max(
        (str(item.get("description") or item.get("message") or "").strip() for item in group),
        key=len,
        default="",
    )

    # Combine distinct remediation guidance & fix versions
    remediations: list[str] = []
    fix_versions: set[str] = set()
    for item in sorted_group:
        rem = str(
            item.get("remediation")
            or item.get("recommendation")
            or item.get("countermeasure")
            or item.get("fix")
            or ""
        ).strip()
        if rem and rem not in remediations:
            remediations.append(rem)
        fix_ver = str(item.get("fixed_version") or item.get("fix_version") or "").strip()
        if fix_ver:
            fix_versions.add(fix_ver)

    merged_remediation = " | ".join(remediations)
    if fix_versions:
        fix_clause = f"Upgrade to fixed version(s): {', '.join(sorted(fix_versions))}."
        if fix_clause not in merged_remediation:
            merged_remediation = (
                f"{merged_remediation} ({fix_clause})".strip()
                if merged_remediation
                else fix_clause
            )

    # Extract CVE & CWE
    cve_id: str | None = None
    cwe_id: str | None = None
    for item in sorted_group:
        if not cve_id:
            cve_id = extract_primary_cve(item)
        if not cwe_id:
            cwe_id = extract_canonical_cwe(item)

    # Determine confidence level
    has_trufflehog_verified = any(
        normalize_engine_name(item.get("engine")) == "trufflehog"
        and (
            bool(item.get("verified"))
            or normalize_confidence(item.get("confidence_level") or item.get("confidence")) == "high"
        )
        for item in group
    )
    if engine_count >= 2 or has_trufflehog_verified or bool(primary.get("is_kev")):
        confidence_level = "high"
    else:
        confidence_level = max(
            (
                normalize_confidence(item.get("confidence_level") or item.get("confidence"))
                for item in group
            ),
            key=lambda c: CONFIDENCE_RANK.get(c, 2),
        )

    category = _classify_category(primary, matched_engines[0])
    file_path = normalize_rel_path(primary.get("file_path") or primary.get("path"))
    line_number = int(
        primary.get("line_number") or primary.get("line") or primary.get("start_line") or 0
    )

    # Build deterministic canonical fingerprint while checking all constituent raw fingerprints
    constituent_fps = {compute_raw_fingerprint(item) for item in group}
    if category == "sca":
        ecosystem, pkg_name, installed_ver = infer_sca_package_coordinates(primary)
        canonical_rule = cve_id or extract_primary_advisory_id(primary)
        line_or_pkg = f"{ecosystem}:{pkg_name}@{installed_ver}"
    elif category == "secret":
        canonical_rule = infer_secret_family(primary)
        line_or_pkg = str(line_number)
    else:
        canonical_rule = cwe_id or str(primary.get("rule_id") or primary.get("title") or "rule")
        line_or_pkg = str(line_number)

    canonical_fp = compute_canonical_fingerprint(
        category=category,
        file_path=file_path,
        canonical_rule_or_cve=canonical_rule,
        line_or_pkg=line_or_pkg,
    )

    # Backward compatibility: if only 1 engine detected the finding and it already had a fingerprint,
    # preserve its original fingerprint unless a suppression matches the canonical fingerprint.
    raw_primary_fp = compute_raw_fingerprint(primary)
    if engine_count == 1 and primary.get("fingerprint"):
        final_fingerprint = str(primary["fingerprint"])
    else:
        final_fingerprint = canonical_fp

    # If ANY constituent raw fingerprint or canonical fingerprint is in FindingSuppression,
    # mark suppressed and align fingerprint with the matched suppression record.
    matched_suppression_fp = next(
        (fp for fp in (constituent_fps | {canonical_fp, raw_primary_fp}) if fp in suppressed_fingerprints),
        None,
    )
    is_suppressed = bool(primary.get("is_suppressed") or primary.get("suppressed")) or (
        matched_suppression_fp is not None
    )
    if matched_suppression_fp:
        final_fingerprint = matched_suppression_fp

    primary["engine"] = matched_engines[0] if engine_count == 1 else ",".join(matched_engines)
    primary["category"] = category
    primary["file_path"] = file_path
    primary["line_number"] = line_number
    primary["severity"] = best_severity
    primary["description"] = best_description or str(primary.get("title") or "")
    if merged_remediation:
        primary["remediation"] = merged_remediation
        primary["countermeasure"] = merged_remediation
    if cve_id:
        primary["cve_id"] = cve_id
    else:
        primary["cve_id"] = None
    if cwe_id and not primary.get("cwe"):
        primary["cwe"] = cwe_id

    primary["matched_engines"] = matched_engines
    primary["matched_engines_json"] = json.dumps(matched_engines)
    primary["engine_count"] = engine_count
    primary["confidence_level"] = confidence_level
    primary["fingerprint"] = final_fingerprint
    primary["constituent_fingerprints"] = sorted(constituent_fps | {canonical_fp})
    primary["is_suppressed"] = is_suppressed
    if is_suppressed:
        primary["status"] = "false_positive"

    return primary


def correlate_findings(
    raw_findings: Iterable[dict[str, Any]],
    suppressed_fingerprints: set[str] | None = None,
) -> CorrelationSummary:
    """
    Entry point invoked by `ScanWorker` after all engines finish and before DB persistence.
    Groups duplicate findings across engines into canonical findings.
    """
    suppressed = suppressed_fingerprints or set()
    raw_list = list(raw_findings)
    buckets: dict[tuple[str, ...], list[dict[str, Any]]] = {}

    for raw in raw_list:
        key = build_correlation_key(raw)
        buckets.setdefault(key, []).append(raw)

    correlated: list[dict[str, Any]] = []
    multi_engine_count = 0

    for corr_key, group in buckets.items():
        merged = _merge_finding_group(corr_key, group, suppressed)
        if merged["engine_count"] >= 2:
            multi_engine_count += 1
        correlated.append(merged)

    return CorrelationSummary(
        raw_findings_count=len(raw_list),
        correlated_findings_count=len(correlated),
        deduplicated_count=max(0, len(raw_list) - len(correlated)),
        multi_engine_confirmed_count=multi_engine_count,
        findings=correlated,
    )
```

```python
def compute_effective_priority_score(
    severity: str | None,
    is_kev: bool = False,
    epss_score: float | None = None,
    engine_count: int = 1,
) -> float:
    """
    Composite Priority Formula on a [0.0, 10.0] scale:
      - Base score by severity: critical=7.0, high=5.0, medium=3.0, low=1.5, info=0.5
      - If `is_kev is True`: severity floor is promoted to `critical` (7.0) and adds +2.5
      - Adds `min(1.5, (epss_score or 0.0) * 2.0)` for exploit probability
      - If `engine_count >= 2`: adds +0.5 for multi-engine confirmation
      - Clamped to [0.0, 10.0] rounded to 2 decimal places
    """
    norm_sev = "critical" if is_kev else normalize_severity(severity)
    base = SEVERITY_BASE_SCORE.get(norm_sev, 3.0)
    kev_boost = 2.5 if is_kev else 0.0
    epss_boost = min(1.5, max(0.0, float(epss_score or 0.0)) * 2.0)
    engine_boost = 0.5 if int(engine_count or 1) >= 2 else 0.0

    raw_score = base + kev_boost + epss_boost + engine_boost
    return round(min(10.0, max(0.0, raw_score)), 2)


def sync_cisa_kev_feed(db: Session, timeout: float = 15.0) -> int:
    """
    Fetches the public CISA Known Exploited Vulnerabilities JSON catalog and upserts
    entries into `CveIntelCache`. Fail-open: returns 0 on network or parse failure.
    """
    try:
        raw_bytes = _http_get_bytes(CISA_KEV_FEED_URL, timeout=timeout)
        payload = json.loads(raw_bytes.decode("utf-8"))
        vulnerabilities = payload.get("vulnerabilities", [])
        if not isinstance(vulnerabilities, list):
            return 0
    except Exception as exc:
        logger.warning("CISA KEV feed sync skipped/failed (fail-open): %s", exc)
        return 0

    now = _utcnow()
    existing_map: dict[str, CveIntelCache] = {
        row.cve_id: row for row in db.query(CveIntelCache).all()
    }

    upserted = 0
    for item in vulnerabilities:
        if not isinstance(item, dict):
            continue
        cve_id = str(item.get("cveID") or "").strip().upper()
        if not cve_id.startswith("CVE-"):
            continue

        row = existing_map.get(cve_id)
        if row is None:
            row = CveIntelCache(cve_id=cve_id)
            db.add(row)
            existing_map[cve_id] = row

        row.is_kev = True
        row.kev_vendor = str(item.get("vendorProject") or "")[:255] or None
        row.kev_product = str(item.get("product") or "")[:255] or None
        row.kev_date_added = str(item.get("dateAdded") or "")[:32] or None
        row.kev_due_date = str(item.get("dueDate") or "")[:32] or None
        row.kev_ransomware_use = str(item.get("knownRansomwareCampaignUse") or "")[:64] or None
        row.updated_at = now
        upserted += 1

    try:
        db.commit()
    except Exception as exc:
        db.rollback()
        logger.warning("Database commit failed during CISA KEV sync: %s", exc)
        return 0

    return upserted


def sync_epss_batch(
    db: Session,
    cve_ids: list[str],
    timeout: float = 5.0,
) -> int:
    """
    Queries the FIRST EPSS REST API in batches of up to 100 CVEs and upserts
    `epss_score` and `epss_percentile` into `CveIntelCache`. Fail-open on any error.
    """
    normalized_cves = sorted(
        {cve.strip().upper() for cve in cve_ids if cve and cve.strip().upper().startswith("CVE-")}
    )
    if not normalized_cves:
        return 0

    now = _utcnow()
    updated_count = 0
    batch_size = 100

    for idx in range(0, len(normalized_cves), batch_size):
        chunk = normalized_cves[idx : idx + batch_size]
        query_param = ",".join(chunk)
        url = f"{FIRST_EPSS_API_URL}?{urllib.parse.urlencode({'cve': query_param})}"

        try:
            raw_bytes = _http_get_bytes(url, timeout=timeout)
            payload = json.loads(raw_bytes.decode("utf-8"))
            data_rows = payload.get("data", [])
            if not isinstance(data_rows, list):
                continue
        except Exception as exc:
            logger.warning("FIRST EPSS batch query failed for %d CVEs (fail-open): %s", len(chunk), exc)
            continue

        existing_rows = {
            row.cve_id: row
            for row in db.query(CveIntelCache).filter(CveIntelCache.cve_id.in_(chunk)).all()
        }

        for entry in data_rows:
            if not isinstance(entry, dict):
                continue
            cve_id = str(entry.get("cve") or "").strip().upper()
            if not cve_id.startswith("CVE-"):
                continue

            try:
                epss_val = float(entry["epss"]) if entry.get("epss") is not None else None
                pct_val = float(entry["percentile"]) if entry.get("percentile") is not None else None
            except (TypeError, ValueError):
                continue

            row = existing_rows.get(cve_id)
            if row is None:
                row = CveIntelCache(cve_id=cve_id, is_kev=False)
                db.add(row)
                existing_rows[cve_id] = row

            row.epss_score = epss_val
            row.epss_percentile = pct_val
            row.updated_at = now
            updated_count += 1

    try:
        db.commit()
    except Exception as exc:
        db.rollback()
        logger.warning("Database commit failed during FIRST EPSS batch sync: %s", exc)
        return 0

    return updated_count


def sync_epss_bulk_csv_for_cached_cves(db: Session, timeout: float = 15.0) -> int:
    """
    Streams `epss_scores-current.csv.gz` and updates EPSS scores for all CVEs currently
    present in `CveIntelCache` (such as KEV entries) without loading 250k rows into memory.
    Falls back to `sync_epss_batch` if the gzip stream is unreachable.
    """
    cached_cves = {row.cve_id: row for row in db.query(CveIntelCache).all()}
    if not cached_cves:
        return 0

    try:
        raw_gz = _http_get_bytes(FIRST_EPSS_BULK_CSV_GZ_URL, timeout=timeout)
        now = _utcnow()
        updated = 0
        with gzip.GzipFile(fileobj=io.BytesIO(raw_gz), mode="rb") as gz_file:
            text_stream = io.TextIOWrapper(gz_file, encoding="utf-8", errors="replace")
            reader = csv.reader(
                line for line in text_stream if line and not line.startswith("#")
            )
            header = next(reader, None)
            if not header:
                return 0
            for parts in reader:
                if len(parts) < 3:
                    continue
                cve_id = parts[0].strip().upper()
                row = cached_cves.get(cve_id)
                if row is None:
                    continue
                try:
                    row.epss_score = float(parts[1])
                    row.epss_percentile = float(parts[2])
                    row.updated_at = now
                    updated += 1
                except ValueError:
                    continue

        db.commit()
        return updated
    except Exception as exc:
        logger.info("Bulk EPSS CSV stream skipped (%s); falling back to batched API sync.", exc)
        return sync_epss_batch(db, list(cached_cves.keys()), timeout=timeout)


def sync_threat_intel_feeds(
    db: Session,
    cve_ids: list[str] | None = None,
    include_bulk_epss: bool = False,

def enrich_findings_with_intel(
    db: Session,
    findings: list[dict[str, Any]],
    auto_fetch_missing_epss: bool = True,
) -> list[dict[str, Any]]:
    """
    Enriches correlated finding dicts in-place with CISA KEV and FIRST EPSS intelligence,
    promotes KEV findings to `critical` severity, computes `effective_priority_score`,
    and sorts findings by `effective_priority_score DESC`.
    """
    if not findings:
        return []

    now = _utcnow()
    stale_cutoff = now - timedelta(hours=24)

    # 1. If the local cache has never synced KEV or is older than 24h, attempt a background KEV sync
    latest_sync = _ensure_aware_utc(db.query(func.max(CveIntelCache.updated_at)).scalar())
    if auto_fetch_missing_epss and (latest_sync is None or latest_sync < stale_cutoff):
        sync_cisa_kev_feed(db, timeout=10.0)

    # 2. Collect all distinct CVE IDs across the scan findings
    distinct_cves: set[str] = set()
    for f in findings:
        cve = f.get("cve_id") or extract_primary_cve(f)
        if isinstance(cve, str) and cve.upper().startswith("CVE-"):
            norm_cve = cve.upper()
            f["cve_id"] = norm_cve
            distinct_cves.add(norm_cve)

    # 3. Query local CveIntelCache
    cache_map: dict[str, CveIntelCache] = {}
    if distinct_cves:
        rows = (
            db.query(CveIntelCache)
            .filter(CveIntelCache.cve_id.in_(sorted(distinct_cves)))
            .all()
        )
        cache_map = {row.cve_id: row for row in rows}

        # 4. Identify missing or stale (>24h) EPSS entries and perform one batched lookup
        if auto_fetch_missing_epss:
            missing_or_stale = [
                cve
                for cve in sorted(distinct_cves)
                if cve not in cache_map
                or cache_map[cve].epss_score is None
                or (
                    _ensure_aware_utc(cache_map[cve].updated_at) is not None
                    and _ensure_aware_utc(cache_map[cve].updated_at) < stale_cutoff
                )
            ]
            if missing_or_stale:
                sync_epss_batch(db, missing_or_stale, timeout=5.0)
                rows = (
                    db.query(CveIntelCache)
                    .filter(CveIntelCache.cve_id.in_(sorted(distinct_cves)))
                    .all()
                )
                cache_map = {row.cve_id: row for row in rows}

    # 5. Attach threat intel fields & calculate effective_priority_score
    for f in findings:
        cve_id = f.get("cve_id")
        intel = cache_map.get(cve_id) if cve_id else None

        is_kev = bool(f.get("is_kev")) or (bool(intel.is_kev) if intel else False)
        kev_due_date = f.get("kev_due_date") or (intel.kev_due_date if intel else None)
        epss_score = (
            float(intel.epss_score)
            if (intel and intel.epss_score is not None)
            else (float(f["epss_score"]) if f.get("epss_score") is not None else None)
        )
        epss_percentile = (
            float(intel.epss_percentile)
            if (intel and intel.epss_percentile is not None)
            else (float(f["epss_percentile"]) if f.get("epss_percentile") is not None else None)
        )

        if is_kev:
            f["severity"] = "critical"
            f["confidence_level"] = "high"

        engine_count = int(f.get("engine_count") or 1)
        priority_score = compute_effective_priority_score(
            severity=f.get("severity"),
            is_kev=is_kev,
            epss_score=epss_score,
            engine_count=engine_count,
        )

        f["is_kev"] = is_kev
        f["kev_due_date"] = kev_due_date
        f["epss_score"] = epss_score
        f["epss_percentile"] = epss_percentile
        f["effective_priority_score"] = priority_score

    # 6. Default sort by effective_priority_score DESC, then severity, then EPSS
    findings.sort(
        key=lambda item: (
            float(item.get("effective_priority_score") or 0.0),
            1 if item.get("is_kev") else 0,
            SEVERITY_SORT_ORDER.get(normalize_severity(item.get("severity")), 0),
            float(item.get("epss_score") or 0.0),
        ),
        reverse=True,
    )
    return findings
```

```python
def classify_license_id(license_str: str | None) -> str:
    """
    Classifies an SPDX license ID or expression into:
      - `restricted_copyleft`
      - `weak_copyleft`
      - `permissive`
      - `unknown`
    """
    if not license_str or not license_str.strip():
        return "unknown"

    normalized = license_str.strip().upper()
    if normalized in ("NONE", "UNKNOWN", "NOASSERTION", "PROPRIETARY"):
        return "unknown"

    # Check weak copyleft BEFORE GPL-* substring checks so LGPL is not misclassified as GPL
    for prefix in WEAK_COPYLEFT_PREFIXES:
        if normalized.startswith(prefix) or f" {prefix}" in normalized:
            return "weak_copyleft"
    if "LESSER GENERAL PUBLIC LICENSE" in normalized or "MOZILLA PUBLIC LICENSE" in normalized:
        return "weak_copyleft"

    for prefix in RESTRICTED_COPYLEFT_PREFIXES:
        if normalized.startswith(prefix) or f" {prefix}" in normalized:
            return "restricted_copyleft"
    if any(
        kw in normalized
        for kw in (
            "AFFERO GENERAL PUBLIC LICENSE",
            "SERVER SIDE PUBLIC LICENSE",
            "GNU GENERAL PUBLIC LICENSE",
            "AGPL",
            "SSPL",
            "GPL",
        )
    ):
        return "restricted_copyleft"

    for prefix in PERMISSIVE_PREFIXES:
        if normalized.startswith(prefix) or f" {prefix}" in normalized:
            return "permissive"
    if any(kw in normalized for kw in ("MIT LICENSE", "APACHE LICENSE", "BSD LICENSE", "ISC LICENSE")):
        return "permissive"

    return "unknown"

                "licenses": licenses_list,
                "properties": [
                    {"name": "aquasecurity:trivy:PkgType", "value": ecosystem},
                    {"name": "veritas:source_file", "value": source_file},
                ],
            }
        )

    # 1. Parse requirements*.txt
    for req_file in sorted(root.rglob("*requirements*.txt")):
        if ".git" in req_file.parts or "node_modules" in req_file.parts:
            continue
        rel_path = str(req_file.relative_to(root)).replace("\\", "/")
        for raw_line in req_file.read_text(encoding="utf-8", errors="replace").splitlines():
            line = raw_line.strip()
            if not line or line.startswith(("#", "-")):
                continue
            # Support inline `# license: AGPL-3.0` annotations in manifests/tests
            inline_license: str | None = None
            if "#" in line:
                code_part, comment_part = line.split("#", 1)
                line = code_part.strip()
                lic_match = re.search(r"license\s*[:=]\s*([A-Za-z0-9.\-+]+)", comment_part, re.I)
                if lic_match:
                    inline_license = lic_match.group(1).strip()

            match = re.match(r"^([A-Za-z0-9_.\-]+)\s*(?:==|>=|~=|<=|>|<)\s*([A-Za-z0-9_.\-+]+)", line)
            if match:
                pkg_name = match.group(1).strip()
                pkg_ver = match.group(2).strip()
            else:
                pkg_name = re.split(r"[;\[\s]", line)[0].strip()
                pkg_ver = "0.0.0"

            if not pkg_name:
                continue
            lic = inline_license or _infer_license_from_metadata_files(root, pkg_name)
            purl = f"pkg:pypi/{pkg_name.lower()}@{pkg_ver}"
            add_component(pkg_name, pkg_ver, "pypi", purl, lic, rel_path)

    # 2. Parse package.json
    for pkg_json in sorted(root.rglob("package.json")):
        if ".git" in pkg_json.parts or "node_modules" in pkg_json.parts:
            continue
        rel_path = str(pkg_json.relative_to(root)).replace("\\", "/")
        try:
            data = json.loads(pkg_json.read_text(encoding="utf-8", errors="replace"))
        except Exception:
            continue
        if not isinstance(data, dict):
            continue

        pkg_licenses_map = data.get("licensePolicies") or data.get("licensesMap") or {}
        for section in ("dependencies", "devDependencies", "optionalDependencies"):
            deps = data.get(section)
            if not isinstance(deps, dict):
                continue
            for dep_name, dep_ver_spec in deps.items():
                if isinstance(dep_ver_spec, dict):
                    clean_ver = str(dep_ver_spec.get("version") or "0.0.0").lstrip("^~>=< ")
                    lic = str(dep_ver_spec.get("license") or "").strip() or None
                else:
                    clean_ver = str(dep_ver_spec or "0.0.0").lstrip("^~>=< ") or "0.0.0"
                    lic = (
                        pkg_licenses_map.get(dep_name)
                        if isinstance(pkg_licenses_map, dict)
                        else None
                    ) or _infer_license_from_metadata_files(root, dep_name)
                purl = f"pkg:npm/{dep_name}@{clean_ver}"
                add_component(dep_name, clean_ver, "npm", purl, lic, rel_path)

    # 3. Parse pyproject.toml
    for pyproject in sorted(root.rglob("pyproject.toml")):
        if ".git" in pyproject.parts or "node_modules" in pyproject.parts:
            continue
        rel_path = str(pyproject.relative_to(root)).replace("\\", "/")
        content = pyproject.read_text(encoding="utf-8", errors="replace")
        for match in re.finditer(
            r'"([A-Za-z0-9_.\-]+)\s*(?:==|>=|~=|<=)\s*([0-9][A-Za-z0-9_.\-]*)"', content
        ):
            pkg_name, pkg_ver = match.group(1), match.group(2)
            lic = _infer_license_from_metadata_files(root, pkg_name)
            purl = f"pkg:pypi/{pkg_name.lower()}@{pkg_ver}"
            add_component(pkg_name, pkg_ver, "pypi", purl, lic, rel_path)

    # 4. Parse go.mod
    for gomod in sorted(root.rglob("go.mod")):
        if ".git" in gomod.parts:
            continue
        rel_path = str(gomod.relative_to(root)).replace("\\", "/")
        for raw_line in gomod.read_text(encoding="utf-8", errors="replace").splitlines():
            line = raw_line.strip()
            if not line or line.startswith(("module ", "go ", "//", "require", ")")):
                continue
            parts = line.split()
            if len(parts) >= 2 and parts[1].startswith("v"):
                pkg_name = parts[0]
                pkg_ver = parts[1]
                lic = _infer_license_from_metadata_files(root, pkg_name)
                purl = f"pkg:golang/{pkg_name}@{pkg_ver}"
                add_component(pkg_name, pkg_ver, "golang", purl, lic, rel_path)

    return {
        "bomFormat": "CycloneDX",
        "specVersion": "1.5",
        "serialNumber": f"urn:uuid:{uuid.uuid4()}",
        "version": 1,
        "metadata": {
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "tools": [
                {
                    "vendor": "VERITAS DevSecOps",
                    "name": "veritas-cyclonedx-generator",
                    "version": "6.0.0",
                }
            ],
        },
        "components": components,
    }


def _extract_component_licenses(comp: dict[str, Any]) -> list[str]:
    licenses_raw = comp.get("licenses")
    extracted: list[str] = []
    if isinstance(licenses_raw, list):
        for entry in licenses_raw:
            if isinstance(entry, dict):
                lic_obj = entry.get("license")
                if isinstance(lic_obj, dict):
                    val = lic_obj.get("id") or lic_obj.get("name")
                    if isinstance(val, str) and val.strip():
                        extracted.append(val.strip())
                elif isinstance(entry.get("expression"), str) and entry["expression"].strip():
                    extracted.append(entry["expression"].strip())
            elif isinstance(entry, str) and entry.strip():
                extracted.append(entry.strip())
    return extracted


def _extract_ecosystem_from_component(comp: dict[str, Any]) -> str:
    purl = str(comp.get("purl") or "")
    if purl.startswith("pkg:"):
        after_pkg = purl[4:]
        eco = after_pkg.split("/", 1)[0].strip().lower()
        if eco:
            return eco
    for prop in comp.get("properties") or []:
        if isinstance(prop, dict) and prop.get("name") in (
            "aquasecurity:trivy:PkgType",
            "veritas:ecosystem",
        ):
            return str(prop.get("value") or "generic").lower()
    return str(comp.get("type") or "library").lower()


def _extract_source_file_from_component(comp: dict[str, Any]) -> str:
    for prop in comp.get("properties") or []:
        if isinstance(prop, dict) and prop.get("name") in (
            "veritas:source_file",
            "aquasecurity:trivy:FilePath",
            "aquasecurity:trivy:SrcFile",
        ):
            val = str(prop.get("value") or "").strip()
            if val:
                return val
    return "SBOM/CycloneDX"


def analyze_cyclonedx_sbom(sbom_data: dict[str, Any]) -> SbomExecutionResult:
    """
    Analyzes a parsed CycloneDX JSON dict, classifies all component licenses,
    builds summary counters, and emits `trivy_license` findings for restricted copyleft packages.
    """
    spec_version_raw = str(sbom_data.get("specVersion") or "1.5")
    spec_version = (
        f"CycloneDX-{spec_version_raw}"
        if not spec_version_raw.startswith("CycloneDX")
        else spec_version_raw
    )

    raw_components = sbom_data.get("components")
    if not isinstance(raw_components, list):
        raw_components = []

    category_counts: dict[str, int] = {
        "permissive": 0,
        "weak_copyleft": 0,
        "restricted_copyleft": 0,
        "unknown": 0,
    }
    by_license: dict[str, int] = {}
    parsed_components: list[dict[str, Any]] = []
    license_findings: list[dict[str, Any]] = []
    policy_violations_count = 0

    tier_priority = {
        "restricted_copyleft": 4,
        "weak_copyleft": 3,
        "unknown": 2,
        "permissive": 1,
    }

    for comp in raw_components:
        if not isinstance(comp, dict):
            continue
        # Skip root application container component entries if present
        if comp.get("type") == "application" and not comp.get("purl"):
            continue

        name = str(comp.get("name") or "unknown").strip()
        version = str(comp.get("version") or "0.0.0").strip()
        purl = str(comp.get("purl") or f"pkg:generic/{name}@{version}").strip()
        ecosystem = _extract_ecosystem_from_component(comp)
        source_file = _extract_source_file_from_component(comp)
        licenses = _extract_component_licenses(comp)

        if not licenses:
            component_tier = "unknown"
            by_license["UNKNOWN"] = by_license.get("UNKNOWN", 0) + 1
        else:
            tiers = [classify_license_id(lic) for lic in licenses]
            component_tier = max(tiers, key=lambda t: tier_priority.get(t, 0))
            for lic in licenses:
                by_license[lic] = by_license.get(lic, 0) + 1

        category_counts[component_tier] = category_counts.get(component_tier, 0) + 1

        parsed_components.append(
            {
                "name": name,
                "version": version,
                "purl": purl,
                "ecosystem": ecosystem,
                "licenses": licenses if licenses else ["UNKNOWN"],
                "risk_tier": component_tier,
                "source_file": source_file,
            }
        )

        if component_tier == "restricted_copyleft":
            policy_violations_count += 1
            restricted_lics = [
                lic for lic in licenses if classify_license_id(lic) == "restricted_copyleft"
            ] or licenses
            lic_display = ", ".join(restricted_lics)
            fp_seed = f"license|{source_file}|{ecosystem}:{name}@{version}|{lic_display}"
            fingerprint = hashlib.sha256(fp_seed.encode("utf-8")).hexdigest()[:32]

            license_findings.append(
                {
                    "engine": "trivy_license",
                    "category": "license",
                    "severity": "high",
                    "cwe": "CWE-1104",
                    "owasp": "A06:2021-Vulnerable and Outdated Components",
                    "rule_id": f"LICENSE-RESTRICTED-{restricted_lics[0].upper()}",
                    "title": f"Restricted Copyleft License Detected: {name}@{version} ({lic_display})",
                    "description": (
                        f"Component '{name}' version '{version}' ({ecosystem}) is distributed under "
                        f"a restricted copyleft license ({lic_display}). Distributing or linking against "
                        f"strong copyleft dependencies in proprietary SaaS or commercial binaries may "
                        f"trigger mandatory source code disclosure obligations."
                    ),
                    "remediation": (
                        f"Replace '{name}@{version}' with a permissive alternative (MIT, Apache-2.0, BSD) "
                        f"or obtain legal approval and commercial dual-licensing."
                    ),
                    "file_path": source_file,
                    "line_number": 1,
                    "package_name": name,
                    "installed_version": version,
                    "ecosystem": ecosystem,
                    "confidence_level": "high",
                    "fingerprint": fingerprint,
                }
            )

    license_summary = {
        "permissive": category_counts["permissive"],
        "weak_copyleft": category_counts["weak_copyleft"],
        "restricted_copyleft": category_counts["restricted_copyleft"],
        "unknown": category_counts["unknown"],
        "by_license": dict(sorted(by_license.items(), key=lambda kv: (-kv[1], kv[0]))),
    }

    sbom_json_str = json.dumps(sbom_data, indent=2)
    return SbomExecutionResult(
        spec_version=spec_version,
        component_count=len(parsed_components),
        policy_violations_count=policy_violations_count,
        license_summary=license_summary,
        components=parsed_components,
        sbom_dict=sbom_data,
        sbom_json=sbom_json_str,
        license_findings=license_findings,
    )


def generate_and_persist_sbom(
    db: Session,
    scan_id: int,
    project_id: int,
    repo_workdir: str | Path,
) -> SbomExecutionResult:
    """
    Runs Trivy in CycloneDX SBOM + license scanner mode on `repo_workdir` (falling back
    to the deterministic manifest parser if `trivy` is not installed or fails),
    persists `ScanSbomArtifact`, and returns the result + any `trivy_license` findings.
    """
    workdir_str = str(repo_workdir)
    sbom_data: dict[str, Any] | None = None

    trivy_bin = shutil.which("trivy")
    if trivy_bin:
        fd, tmp_sbom_path = tempfile.mkstemp(prefix=f"veritas-sbom-{scan_id}-", suffix=".cdx.json")
        os.close(fd)
        try:
            cmd = [
                trivy_bin,
                "fs",
                "--format",
                "cyclonedx",
                "--scanners",
                "license",
                "--quiet",
                "--output",
                tmp_sbom_path,
                workdir_str,
            ]
            proc = subprocess.run(
                cmd,
                shell=False,
                capture_output=True,
                text=True,
                timeout=180,
                check=False,
            )
            if proc.returncode == 0 and os.path.exists(tmp_sbom_path) and os.path.getsize(tmp_sbom_path) > 0:
                with open(tmp_sbom_path, "r", encoding="utf-8", errors="replace") as f:
                    parsed = json.load(f)
                if isinstance(parsed, dict) and parsed.get("bomFormat") == "CycloneDX":
                    sbom_data = parsed
        except Exception as exc:
            logger.warning("Trivy CycloneDX CLI execution failed; using fallback manifest parser: %s", exc)
        finally:
            try:
                if os.path.exists(tmp_sbom_path):
                    os.remove(tmp_sbom_path)
            except OSError:
                pass

    if sbom_data is None:
        sbom_data = _parse_manifests_fallback(workdir_str)

    result = analyze_cyclonedx_sbom(sbom_data)

    # Upsert ScanSbomArtifact record for this scan
    existing = (
        db.query(ScanSbomArtifact)
        .filter(ScanSbomArtifact.scan_id == scan_id)
        .first()
    )
    if existing is None:
        artifact = ScanSbomArtifact(
            scan_id=scan_id,
            project_id=project_id,
            spec_version=result.spec_version,
            component_count=result.component_count,
            license_summary_json=json.dumps(result.license_summary),
            policy_violations_count=result.policy_violations_count,
            sbom_json=result.sbom_json,
        )
        db.add(artifact)
    else:
        existing.spec_version = result.spec_version
        existing.component_count = result.component_count
        existing.license_summary_json = json.dumps(result.license_summary)
        existing.policy_violations_count = result.policy_violations_count
        existing.sbom_json = result.sbom_json

    db.commit()
    return result
```

```python
def execute_phase6a_post_processing(
    db: Session,
    scan: Scan,
    repo_workdir: str,
    raw_engine_findings: list[dict],
    auto_fetch_missing_epss: bool = True,
) -> tuple[ list[Finding], dict ]:
    """
    Invoked by ScanWorker after all 12 security engines finish scanning `repo_workdir`:
      1. Generates & persists the CycloneDX 1.5 SBOM and appends any restricted copyleft
         `trivy_license` findings to `raw_engine_findings`.
      2. Correlates & deduplicates findings across all engines while honoring `FindingSuppression`.
      3. Enriches CVE findings with CISA KEV and FIRST EPSS scores and computes `effective_priority_score`.
      4. Persists canonical `Finding` rows sorted by priority and returns the Groq executive context.
    """
    # Step 1: CycloneDX SBOM + License Governance
    sbom_result = generate_and_persist_sbom(
        db=db,
        scan_id=scan.id,
        project_id=scan.project_id,
        repo_workdir=repo_workdir,
    )
    combined_raw_findings = list(raw_engine_findings) + list(sbom_result.license_findings)

    # Step 2: Load active project suppressions
    suppressed_fps = {
        row.fingerprint
        for row in db.query(FindingSuppression)
        .filter(FindingSuppression.project_id == scan.project_id)
        .all()
        if row.fingerprint
    }

    # Step 3: Cross-Engine Canonical Correlation
    correlation_summary = correlate_findings(
        raw_findings=combined_raw_findings,
        suppressed_fingerprints=suppressed_fps,
    )

    # Step 4: Offline CISA KEV + FIRST EPSS Enrichment & Priority Scoring
    enriched_dicts = enrich_findings_with_intel(
        db=db,
        findings=correlation_summary.findings,
        auto_fetch_missing_epss=auto_fetch_missing_epss,
    )

    # Step 5: Persist canonical Finding ORM objects
    persisted_findings: list[Finding] = []
    for item in enriched_dicts:
        matched_engines = item.get("matched_engines") or [item.get("engine") or "unknown"]
        finding_row = Finding(
            scan_id=scan.id,
            project_id=scan.project_id,
            engine=str(item.get("engine") or matched_engines[0]),
            category=str(item.get("category") or "sast"),
            severity=str(item.get("severity") or "medium"),
            title=str(item.get("title") or "Security Finding"),
            description=str(item.get("description") or ""),
            file_path=str(item.get("file_path") or "unknown"),
            line_number=int(item.get("line_number") or 0),
            cwe=item.get("cwe"),
            owasp=item.get("owasp"),
            code_snippet=item.get("code_snippet"),
            countermeasure=item.get("remediation") or item.get("countermeasure"),
            fingerprint=str(item.get("fingerprint") or ""),
            status="false_positive" if item.get("is_suppressed") else str(item.get("status") or "open"),
            is_suppressed=bool(item.get("is_suppressed", False)),
            # Phase 6A fields:
            cve_id=item.get("cve_id"),
            matched_engines_json=json.dumps(matched_engines),
            engine_count=int(item.get("engine_count") or len(matched_engines)),
            confidence_level=str(item.get("confidence_level") or "medium"),
            epss_score=item.get("epss_score"),
            epss_percentile=item.get("epss_percentile"),
            is_kev=bool(item.get("is_kev", False)),
            kev_due_date=item.get("kev_due_date"),
            effective_priority_score=float(item.get("effective_priority_score") or 0.0),
        )
        db.add(finding_row)
        persisted_findings.append(finding_row)

    # Update Scan correlation counters if present on Scan model
    if hasattr(scan, "raw_findings_count"):
        scan.raw_findings_count = correlation_summary.raw_findings_count
    if hasattr(scan, "correlated_findings_count"):
        scan.correlated_findings_count = correlation_summary.correlated_findings_count
    if hasattr(scan, "kev_findings_count"):
        scan.kev_findings_count = sum(1 for f in enriched_dicts if f.get("is_kev"))

    db.commit()
    for f_row in persisted_findings:
        db.refresh(f_row)

    # Step 6: Build Executive Intelligence Summary for Groq (`services/ai_report.py`)
    exec_intel_context = build_executive_intel_context(
        raw_findings_count=correlation_summary.raw_findings_count,
        correlated_findings=enriched_dicts,
    )
    return persisted_findings, exec_intel_context
```

```python
    scan_id: int,
    is_kev: bool | None = Query(default=None),
    min_epss: float | None = Query(default=None, ge=0.0, le=1.0),

        query = query.order_by(
            desc(severity_rank_expr),
            desc(Finding.effective_priority_score),
            Finding.id.asc(),
        )
    elif sort_by == "epss":
        query = query.order_by(
            desc(case((Finding.epss_score.is_(None), -1.0), else_=Finding.epss_score)),
            desc(Finding.effective_priority_score),
            Finding.id.asc(),
        )
    else:
        # Default: sort_by == "priority"
        query = query.order_by(
            desc(Finding.effective_priority_score),
            desc(Finding.is_kev),
            desc(severity_rank_expr),
            Finding.id.asc(),
        )

    rows = query.all()
    serialized = [serialize_finding_phase6a(row) for row in rows]

    return {
        "scan_id": scan.id,
        "project_id": project_id,
        "raw_findings_count": getattr(scan, "raw_findings_count", len(serialized)) or len(serialized),
        "correlated_findings_count": getattr(scan, "correlated_findings_count", len(serialized)) or len(serialized),
        "total": len(serialized),
        "findings": serialized,
    }
```
