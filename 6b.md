    """Idempotent Phase 6B schema migrations for SQLite and PostgreSQL."""
    Base.metadata.create_all(bind=db.engine)

    tables = set(inspector.get_table_names())

    with db.engine.begin() as connection:
        if "projects" in tables:
            proj_cols = {c["name"] for c in inspector.get_columns("projects")}
            proj_alters = {
                "sla_critical_days": "ALTER TABLE projects ADD COLUMN sla_critical_days INTEGER DEFAULT 7",
                "sla_high_days": "ALTER TABLE projects ADD COLUMN sla_high_days INTEGER DEFAULT 14",
                "sla_medium_days": "ALTER TABLE projects ADD COLUMN sla_medium_days INTEGER DEFAULT 30",
                "sla_low_days": "ALTER TABLE projects ADD COLUMN sla_low_days INTEGER DEFAULT 90",
                "require_suppression_approval": "ALTER TABLE projects ADD COLUMN require_suppression_approval BOOLEAN DEFAULT 0",
            }
            for col_name, ddl in proj_alters.items():
                if col_name not in proj_cols:

def compute_finding_sla(
    project: Project,
    *,
    severity: str | None,
    is_kev: bool = False,
    base_time=None,
) -> tuple[Any, bool]:
    """
    Computes `(sla_due_at, sla_breached)` for a finding based on the project's SLA policy.
    """
    now = utcnow()
    ref_time = ensure_utc(base_time) if base_time else now
    sev = "critical" if is_kev else (severity or "info").lower()

    if sev == "critical":
        days = int(getattr(project, "sla_critical_days", 7) or 7)
    elif sev == "high":
        days = int(getattr(project, "sla_high_days", 14) or 14)
    elif sev == "medium":
        days = int(getattr(project, "sla_medium_days", 30) or 30)
    else:
        days = int(getattr(project, "sla_low_days", 90) or 90)

    due_at = ref_time + timedelta(days=max(1, days))
    breached = now > due_at
    return due_at, breached


def _extract_cwe_number(cwe_str: str | None) -> int | None:
    if not cwe_str:
        return None
    m = CWE_NUM_REGEX.search(str(cwe_str))
    return int(m.group(1)) if m else None


def map_finding_to_owasp_ids(f: Finding) -> set[str]:
    matched: set[str] = set()
    owasp_cat = (f.owasp_category or "").upper()
    for ctrl in OWASP_TOP_10_CONTROLS:
        cid = ctrl["id"]
        if cid in owasp_cat or cid.split(":")[0] in owasp_cat:
            matched.add(cid)

    cwe_num = _extract_cwe_number(f.cwe)
    fam = (f.vuln_family or "").lower()
    eng = (f.engine or "").lower()

    if fam == "license" or eng == "trivy_license" or cwe_num == 1104:
        matched.add("A06:2021")
    if fam in ("sca", "dependency") or eng in ("osv", "pip_audit") or getattr(f, "cve_id", None) or getattr(f, "is_kev", False):
        matched.add("A06:2021")
    if fam in ("secret", "secrets") or eng in ("gitleaks", "trufflehog", "detect_secrets"):
        matched.add("A07:2021")
    if eng in ("checkov", "hadolint") or fam in ("iac", "container", "dockerfile"):
        matched.add("A05:2021")
    if fam == "sqli" or fam == "xss":
        matched.add("A03:2021")
    if fam == "csrf":
        matched.add("A01:2021")

    for ctrl in OWASP_TOP_10_CONTROLS:
        if cwe_num and cwe_num in ctrl["cwes"]:
            matched.add(ctrl["id"])
        if fam and fam in ctrl["families"]:
            matched.add(ctrl["id"])

    if not matched:
        matched.add("A04:2021")

    return matched


def _score_controls(
    controls_spec: list[dict[str, Any]],
    findings_by_control: dict[str, list[Finding]],
) -> dict[str, Any]:
    evaluated_controls: list[dict[str, Any]] = []
    total_points = 0.0
    pass_count = 0
    warn_count = 0
    fail_count = 0

    for spec in controls_spec:
        cid = spec["id"]
        matched = findings_by_control.get(cid, [])
        counts = {"critical": 0, "high": 0, "medium": 0, "low": 0, "info": 0}
        has_kev = False
        for f in matched:
            sev = (f.severity or "info").lower()
            counts[sev] = counts.get(sev, 0) + 1
            if getattr(f, "is_kev", False):
                has_kev = True

        if counts["critical"] > 0 or counts["high"] > 0 or has_kev:
            status = "fail"
            fail_count += 1
            points = 0.0
        elif counts["medium"] > 0 or counts["low"] > 0 or counts["info"] > 0:
            status = "warn"
            warn_count += 1
            points = 0.6
        else:
            status = "pass"
            pass_count += 1
            points = 1.0

        total_points += points
        evaluated_controls.append(
            {
                "control_id": cid,
                "title": spec["title"],
                "description": spec["description"],
                "status": status,
                "finding_count": len(matched),
                "finding_counts": counts,
                "sample_findings": [
                    {
                        "id": f.id,
                        "title": f.title,
                        "severity": f.severity,
                        "cwe": f.cwe,
                        "cve_id": getattr(f, "cve_id", None),
                        "is_kev": bool(getattr(f, "is_kev", False)),
                        "file_path": f.file_path,
                    }
                    for f in matched[:5]
                ],
            }
        )

    total_controls = len(controls_spec) or 1
    readiness_score = round((total_points / total_controls) * 100.0, 1)
    if fail_count == 0 and readiness_score >= 85.0:
        verdict = "Audit Ready"
    elif fail_count <= 2 and readiness_score >= 60.0:
        verdict = "Needs Remediation"
    else:
        verdict = "Critical Non-Compliance"

    return {
        "readiness_score": readiness_score,
        "verdict": verdict,
        "pass_count": pass_count,
        "warn_count": warn_count,
        "fail_count": fail_count,
        "total_controls": len(controls_spec),
        "controls": evaluated_controls,
    }


def evaluate_scan_compliance(
    db: Session,
    project: Project,
    scan: Scan,
) -> dict[str, Any]:
    """
    Evaluates a scan's open findings and SBOM artifact against OWASP Top 10:2021,
    SOC 2 Type II, and ISO/IEC 27001:2022 Annex A, plus SLA & suppression metrics.
    """
    all_findings = list(db.scalars(select(Finding).where(Finding.scan_id == scan.id)).all())
    active_findings = [f for f in all_findings if f.status in ("open", "triage")]

    now = utcnow()
    overdue_sla_count = 0
    kev_open_count = 0

    for f in active_findings:
        if getattr(f, "is_kev", False):
            kev_open_count += 1
        due = ensure_utc(f.sla_due_at) if getattr(f, "sla_due_at", None) else None
        if due is None:
            due, breached = compute_finding_sla(
                project,
                severity=f.severity,
                is_kev=bool(getattr(f, "is_kev", False)),
                base_time=f.created_at,
            )
        else:
            breached = now > due
        if breached:
            overdue_sla_count += 1

    sbom_artifact = db.scalar(
        select(ScanSbomArtifact).where(ScanSbomArtifact.scan_id == scan.id)
    )
    pending_suppressions_count = len(
        list(
            db.scalars(
                select(FindingSuppression).where(
                    FindingSuppression.project_id == project.id,
                    FindingSuppression.status == "pending_approval",
                )
            ).all()
        )
    )

    owasp_buckets: dict[str, list[Finding]] = {c["id"]: [] for c in OWASP_TOP_10_CONTROLS}
    for f in active_findings:
        for owasp_id in map_finding_to_owasp_ids(f):
            owasp_buckets.setdefault(owasp_id, []).append(f)

    soc2_buckets: dict[str, list[Finding]] = {c["id"]: [] for c in SOC2_CONTROLS}
    for spec in SOC2_CONTROLS:
        cid = spec["id"]
        seen_ids: set[str] = set()
        for owasp_ref in spec.get("owasp_refs", set()):
            for f in owasp_buckets.get(owasp_ref, []):
                if f.id not in seen_ids:
                    seen_ids.add(f.id)
                    soc2_buckets[cid].append(f)
        if spec.get("match_kev_or_sla"):
            for f in active_findings:
                if (getattr(f, "is_kev", False) or getattr(f, "sla_breached", False)) and f.id not in seen_ids:
                    seen_ids.add(f.id)
                    soc2_buckets[cid].append(f)

    iso_buckets: dict[str, list[Finding]] = {c["id"]: [] for c in ISO27001_CONTROLS}
    for spec in ISO27001_CONTROLS:
        cid = spec["id"]
        seen_ids = set()
        if spec.get("match_license_only"):
            for f in active_findings:
                cwe_num = _extract_cwe_number(f.cwe)
                if f.vuln_family == "license" or f.engine == "trivy_license" or cwe_num == 1104:
                    if f.id not in seen_ids:
                        seen_ids.add(f.id)
                        iso_buckets[cid].append(f)
            continue

        for owasp_ref in spec.get("owasp_refs", set()):
            for f in owasp_buckets.get(owasp_ref, []):
                if f.id not in seen_ids:
                    seen_ids.add(f.id)
                    iso_buckets[cid].append(f)
        if spec.get("match_kev_or_sla"):
            for f in active_findings:
                if (getattr(f, "is_kev", False) or getattr(f, "sla_breached", False)) and f.id not in seen_ids:
                    seen_ids.add(f.id)
                    iso_buckets[cid].append(f)

    owasp_eval = _score_controls(OWASP_TOP_10_CONTROLS, owasp_buckets)
    soc2_eval = _score_controls(SOC2_CONTROLS, soc2_buckets)
    iso_eval = _score_controls(ISO27001_CONTROLS, iso_buckets)

    overall_score = round(
        (owasp_eval["readiness_score"] + soc2_eval["readiness_score"] + iso_eval["readiness_score"]) / 3.0,
        1,
    )
    if overall_score >= 85.0 and kev_open_count == 0:
        overall_verdict = "Audit Ready"
    elif overall_score >= 60.0 and kev_open_count == 0:
        overall_verdict = "Needs Remediation"
    else:
        overall_verdict = "Critical Non-Compliance"

    return {
        "scan_id": scan.id,
        "project_id": project.id,
        "project_name": project.name,
        "target": scan.target,
        "commit_sha": scan.commit_sha,
        "evaluated_at": now.isoformat(),
        "overall_readiness_score": overall_score,
        "overall_verdict": overall_verdict,
        "total_open_findings": len(active_findings),
        "total_suppressed_findings": len(all_findings) - len(active_findings),
        "kev_open_count": kev_open_count,
        "total_overdue_sla": overdue_sla_count,
        "pending_suppressions_count": pending_suppressions_count,
        "sbom_verified": sbom_artifact is not None,
        "sbom_component_count": sbom_artifact.component_count if sbom_artifact else 0,
        "sbom_policy_violations": sbom_artifact.policy_violations_count if sbom_artifact else 0,
        "frameworks": {
            "owasp_top_10": {
                "name": "OWASP Top 10 (2021)",
                **owasp_eval,
            },
            "soc2": {
                "name": "SOC 2 Type II (Trust Services Criteria)",
                **soc2_eval,
            },
            "iso27001": {
                "name": "ISO/IEC 27001:2022 Annex A",
                **iso_eval,
            },
        },
    }

def expire_stale_suppressions(db: Session, project_id: str) -> int:
    """
    Transitions any `FindingSuppression` with `expires_at < utcnow()` to `status='expired'`
    and re-opens any findings in the project's scans that were suppressed solely by that fingerprint.
    """
    now = utcnow()
    rows = list(
        db.scalars(
            select(FindingSuppression).where(
                FindingSuppression.project_id == project_id,
                FindingSuppression.status == "approved",
                FindingSuppression.expires_at.isnot(None),
            )
        ).all()
    )
    expired_fps: set[str] = set()
    for row in rows:
        exp = ensure_utc(row.expires_at)
        if exp and exp < now:
            row.status = "expired"
            db.add(row)
            if row.fingerprint:
                expired_fps.add(row.fingerprint)

    if expired_fps:
        scan_ids = list(
            db.scalars(select(Scan.id).where(Scan.project_id == project_id)).all()
        )
        if scan_ids:
            findings_to_reopen = list(
                db.scalars(
                    select(Finding).where(
                        Finding.scan_id.in_(scan_ids),
                        Finding.fingerprint.in_(sorted(expired_fps)),
                        Finding.status == "false_positive",
                    )
                ).all()
            )
            for f in findings_to_reopen:
                f.status = "open"
                db.add(f)
        db.commit()

    return len(expired_fps)

def public_finding(finding: Finding) -> dict:
    countermeasures = None
    if finding.countermeasures_json:
        try:
            countermeasures = json.loads(finding.countermeasures_json)
        except json.JSONDecodeError:
            countermeasures = None

    matched_engines: list[str] = []
    raw_matched = getattr(finding, "matched_engines_json", None)
    if raw_matched:
        try:
            parsed = json.loads(raw_matched)
            if isinstance(parsed, list):
                matched_engines = [str(e) for e in parsed if e]
        except json.JSONDecodeError:
            matched_engines = []
    if not matched_engines and finding.engine:
        matched_engines = [finding.engine]

    engine_count = int(getattr(finding, "engine_count", 0) or len(matched_engines) or 1)

def update_finding_status(
    db: Session,
    user: User,
    finding_id: str,
    *,
    status: str,
    suppress: bool = False,
    reason: str | None = None,
    suppression_type: str = "false_positive",
    justification: str | None = None,
    expires_in_days: int | None = None,
    request_approval: bool = False,
) -> Finding:
    if status not in VALID_FINDING_STATUS:
        raise HTTPException(status_code=400, detail="Invalid finding status.")
    finding = db.scalar(select(Finding).where(Finding.id == finding_id))
    if not finding:
        raise HTTPException(status_code=404, detail="Finding not found.")

    scan, project = get_owned_scan(db, user, finding.scan_id, require_write=True)
    role = get_project_role(db, user, project) or "developer"

    if not finding.fingerprint:
        finding.fingerprint = finding_fingerprint(
            engine=finding.engine,
            rule_id=finding.rule_id,
            file_path=finding.file_path,
            line_start=finding.line_start,
            title=finding.title,
        )

    wants_suppression = suppress or status == "false_positive"
    needs_two_person_review = wants_suppression and (
        role == "developer"
        or request_approval
        or (bool(getattr(project, "require_suppression_approval", False)) and role not in {"owner", "security_lead"})
    )

    expires_at = None
    if expires_in_days is not None and int(expires_in_days) > 0:
        expires_at = utcnow() + timedelta(days=int(expires_in_days))

    if wants_suppression and finding.fingerprint:
        existing = db.scalar(
            select(FindingSuppression).where(
                FindingSuppression.project_id == project.id,
                FindingSuppression.fingerprint == finding.fingerprint,
            )
        )
        effective_reason = (justification or reason or "")[:512] or None
        effective_just = justification or reason or None

        if needs_two_person_review:
            if existing is None:
                supp_row = FindingSuppression(
                    project_id=project.id,
                    fingerprint=finding.fingerprint,
                    reason=effective_reason,
                    created_by_user_id=user.id,
                    status="pending_approval",
                    suppression_type=suppression_type or "risk_accepted",
                    justification=effective_just,
                    requested_by_user_id=user.id,
                    expires_at=expires_at,
                )
                db.add(supp_row)
            else:
                existing.status = "pending_approval"
                existing.suppression_type = suppression_type or existing.suppression_type
                existing.reason = effective_reason
                existing.justification = effective_just
                existing.requested_by_user_id = user.id
                existing.expires_at = expires_at
                db.add(existing)
            # Finding remains open/triage until approved by security_lead or owner
            if finding.status == "false_positive":
                finding.status = "open"
        else:
            now = utcnow()
            if existing is None:
                supp_row = FindingSuppression(
                    project_id=project.id,
                    fingerprint=finding.fingerprint,
                    reason=effective_reason,
                    created_by_user_id=user.id,
                    status="approved",
                    suppression_type=suppression_type or "false_positive",
                    justification=effective_just,
                    requested_by_user_id=user.id,
                    reviewed_by_user_id=user.id,
                    reviewed_at=now,
                    expires_at=expires_at,
                )
                db.add(supp_row)
            else:
                existing.status = "approved"
                existing.suppression_type = suppression_type or existing.suppression_type
                existing.reason = effective_reason
                existing.justification = effective_just
                existing.reviewed_by_user_id = user.id
                existing.reviewed_at = now
                existing.expires_at = expires_at
                db.add(existing)
            finding.status = status
    else:
        finding.status = status

    db.add(finding)
    db.commit()
    db.refresh(finding)
    return finding

            added_by_user_id=actor.id,

def review_finding_suppression(
    db: Session,
    reviewer: User,
    project_id: str,
    suppression_id: str,
    *,
    decision: str,
    review_note: str | None = None,
    expires_in_days: int | None = None,
) -> dict:
    project, role = get_project_with_role(db, reviewer, project_id)
    if role not in {"owner", "security_lead"} and reviewer.role != "admin":
        raise HTTPException(
            status_code=403,
            detail="Only project owners, security leads, or admins can review suppressions.",
        )

    norm_decision = (decision or "").strip().lower()
    if norm_decision not in {"approved", "rejected"}:
        raise HTTPException(status_code=400, detail="Decision must be 'approved' or 'rejected'.")

    supp = db.scalar(
        select(FindingSuppression).where(
            FindingSuppression.id == suppression_id,
            FindingSuppression.project_id == project.id,
        )
    )
    if not supp:
        raise HTTPException(status_code=404, detail="Suppression record not found.")

    now = utcnow()
    supp.status = norm_decision
    supp.reviewed_by_user_id = reviewer.id
    supp.reviewed_at = now
    if review_note:
        supp.reason = review_note[:512]
    if expires_in_days is not None and int(expires_in_days) > 0:
        supp.expires_at = now + timedelta(days=int(expires_in_days))

    db.add(supp)

    # Apply decision across project scans matching this fingerprint
    scan_ids = list(db.scalars(select(Scan.id).where(Scan.project_id == project.id)).all())
    if scan_ids and supp.fingerprint:
        matching_findings = list(
            db.scalars(
                select(Finding).where(
                    Finding.scan_id.in_(scan_ids),
                    Finding.fingerprint == supp.fingerprint,
                )
            ).all()
        )
        for f in matching_findings:
            if norm_decision == "approved":
                f.status = "false_positive"
            elif norm_decision == "rejected" and f.status == "false_positive":
                f.status = "open"
            db.add(f)

    record_audit_event(
        db,
        action="finding.suppression.reviewed",
        actor_user_id=reviewer.id,
        target_user_id=supp.requested_by_user_id or supp.created_by_user_id,
        metadata={
            "project_id": project.id,
            "suppression_id": supp.id,
            "fingerprint": supp.fingerprint,
            "decision": norm_decision,
        },
    )
    db.commit()
    db.refresh(supp)
    return public_suppression(supp)
