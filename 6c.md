from __future__ import annotations

import ipaddress
import json
import logging
import os
import re
import socket
import ssl
from datetime import datetime, timezone
from typing import Any
from urllib.parse import urljoin, urlparse

import httpx

from app.services.engines.normalize import NormalizedFinding

logger = logging.getLogger(__name__)

VERSION_DISCLOSURE_RE = re.compile(r"[A-Za-z0-9_\-]+/\d+\.\d+(?:\.\d+)?|PHP/\d|Express", re.IGNORECASE)
ENV_KEY_VALUE_RE = re.compile(r"^[A-Z][A-Z0-9_]{2,}=\S+", re.MULTILINE)


class DastTargetBlockedError(ValueError):
    """Raised when a DAST target URL points to a restricted loopback/private/metadata IP."""


def _allow_local_dast() -> bool:
    return (
        os.getenv("VERITAS_ALLOW_LOCAL_DAST", "").strip() in ("1", "true", "yes")
        or os.getenv("APP_ENV", "").strip().lower() == "test"
    )


def validate_dast_target_url(url: str, *, enforce_public_ip: bool | None = None) -> str:
    """
    Validates that `url` uses `http://` or `https://` and does not resolve to
    loopback, link-local, RFC1918 private, or cloud metadata IP addresses unless
    explicitly permitted via `VERITAS_ALLOW_LOCAL_DAST=1`.
    """
    cleaned = (url or "").strip()
    parsed = urlparse(cleaned)
    if parsed.scheme.lower() not in ("http", "https") or not parsed.hostname:
        raise DastTargetBlockedError("DAST target must be a valid http:// or https:// URL.")

    hostname = parsed.hostname.strip().lower()
    must_enforce = (not _allow_local_dast()) if enforce_public_ip is None else enforce_public_ip

    if must_enforce:
        if hostname in ("localhost", "localhost.localdomain", "metadata.google.internal"):
            raise DastTargetBlockedError(f"DAST target host '{hostname}' is blocked by SSRF policy.")

        # Check literal IP first
        try:
            ip_obj = ipaddress.ip_address(hostname)
            if (
                ip_obj.is_private
                or ip_obj.is_loopback
                or ip_obj.is_link_local
                or ip_obj.is_multicast
                or ip_obj.is_reserved
                or ip_obj.is_unspecified
            ):
                raise DastTargetBlockedError(
                    f"DAST target IP '{ip_obj}' is restricted (private/loopback/metadata range)."
                )
        except ValueError:
            # Resolve DNS and inspect all returned A/AAAA records
            try:
                addr_info = socket.getaddrinfo(hostname, parsed.port or (443 if parsed.scheme == "https" else 80))
            except socket.gaierror as exc:
                raise DastTargetBlockedError(f"Could not resolve DAST target host '{hostname}': {exc}") from exc

            for family, _, _, _, sockaddr in addr_info:
                if family not in (socket.AF_INET, socket.AF_INET6):
                    continue
                ip_str = sockaddr[0]
                ip_obj = ipaddress.ip_address(ip_str)
                if (
                    ip_obj.is_private
                    or ip_obj.is_loopback
                    or ip_obj.is_link_local
                    or ip_obj.is_multicast
                    or ip_obj.is_reserved
                    or ip_obj.is_unspecified
                ):
                    raise DastTargetBlockedError(
                        f"DAST target '{hostname}' resolved to restricted IP '{ip_obj}'."
                    )

    return cleaned


def _inspect_tls_endpoint(hostname: str, port: int, timeout: float = 5.0) -> list[NormalizedFinding]:
    findings: list[NormalizedFinding] = []
    ctx = ssl.create_default_context()
    try:
        with socket.create_connection((hostname, port), timeout=timeout) as sock:
            with ctx.wrap_socket(sock, server_hostname=hostname) as ssock:
                tls_ver = ssock.version() or ""
                if tls_ver in ("SSLv2", "SSLv3", "TLSv1", "TLSv1.1"):
                    findings.append(
                        NormalizedFinding(
                            engine="veritas_dast",
                            title=f"Deprecated TLS Protocol Negotiated ({tls_ver})",
                            severity="high",
                            rule_id="dast.tls.deprecated_version",
                            vuln_family="crypto",
                            cwe="CWE-326",
                            owasp_category="A02:2021-Cryptographic Failures",
                            message=f"Endpoint {hostname}:{port} negotiated legacy protocol {tls_ver}. Enforce TLS 1.2 or TLS 1.3.",
                            file_path=f"https://{hostname}:{port}",
                            raw={"tls_version": tls_ver, "confidence_level": "high"},
                        )
                    )
                cert = ssock.getpeercert()
                not_after = cert.get("notAfter") if isinstance(cert, dict) else None
                if isinstance(not_after, str):
                    exp_ts = ssl.cert_time_to_seconds(not_after)
                    days_left = (exp_ts - datetime.now(timezone.utc).timestamp()) / 86400.0
                    if days_left < 14:
                        findings.append(
                            NormalizedFinding(
                                engine="veritas_dast",
                                title=f"TLS Certificate Expiring Soon ({int(days_left)} days remaining)",
                                severity="medium" if days_left > 0 else "critical",
                                rule_id="dast.tls.cert_expiring",
                                vuln_family="crypto",
                                cwe="CWE-295",
                                owasp_category="A02:2021-Cryptographic Failures",
                                message=f"TLS certificate for {hostname} expires on {not_after} ({int(days_left)} days remaining).",
                                file_path=f"https://{hostname}:{port}",
                                raw={"days_remaining": round(days_left, 1), "confidence_level": "high"},
                            )
                        )
    except Exception:
        # In unit test mocks or custom certs without live TCP socket, skip socket-level TLS check gracefully
        return findings
    return findings


def _ looks_like_html_fallback(text: str, content_type: str) -> bool:
    ct_lower = (content_type or "").lower()
    head = (text or "").lstrip()[:300].lower()
    return (
        "text/html" in ct_lower
        or head.startswith("<!doctype html")
        or head.startswith("<html")
        or "<div id=\"root\"" in head
    )


def run_dast_probe(
    target_url: str,
    *,
    security_level: str = "standard",
    transport: httpx.BaseTransport | None = None,
    enforce_public_ip: bool | None = None,
) -> tuple[list[NormalizedFinding], dict[str, Any]]:
    """
    Executes non-destructive live web vulnerability checks (TLS, HTTP security headers,
    cookie security flags, CORS origin reflection, and sensitive path disclosures).
    Returns `(findings, engine_meta)`.
    """
    meta: dict[str, Any] = {
        "engine": "veritas_dast",
        "available": True,
        "target_url": target_url,
        "findings": 0,
    }
    findings: list[NormalizedFinding] = []

    try:
        validated_url = validate_dast_target_url(target_url, enforce_public_ip=enforce_public_ip)
    except DastTargetBlockedError as exc:
        meta["skipped"] = str(exc)
        return [], meta

    parsed = urlparse(validated_url)
    is_https = parsed.scheme.lower() == "https"

    if is_https and parsed.hostname and transport is None and not _allow_local_dast():
        findings.extend(_inspect_tls_endpoint(parsed.hostname, parsed.port or 443))

    client_kwargs: dict[str, Any] = {
        "follow_redirects": True,
        "timeout": 10.0,
        "headers": {"User-Agent": "VERITAS-DAST-Scanner/2.1 (+https://veritas.trackifyapp.co.in)"},
    }
    if transport is not None:
        client_kwargs["transport"] = transport

    try:
        with httpx.Client(**client_kwargs) as client:
            resp = client.get(validated_url)
            final_url = str(resp.url)
            final_scheme = urlparse(final_url).scheme.lower()
            headers = resp.headers

            # 1. Cleartext HTTP transport check
            if final_scheme == "http":
                findings.append(
                    NormalizedFinding(
                        engine="veritas_dast",
                        title="Cleartext HTTP Transport Without HTTPS Enforcement",
                        severity="high",
                        rule_id="dast.http.cleartext",
                        vuln_family="crypto",
                        cwe="CWE-319",
                        owasp_category="A02:2021-Cryptographic Failures",
                        message=(
                            f"Target {validated_url} serves traffic over unencrypted HTTP ({final_url}) "
                            "without redirecting to HTTPS, exposing credentials and session tokens in transit."
                        ),
                        file_path=final_url,
                        raw={"url": final_url, "confidence_level": "high"},
                    )
                )

            # 2. Content-Security-Policy check
            csp = headers.get("content-security-policy")
            if not csp:
                findings.append(
                    NormalizedFinding(
                        engine="veritas_dast",
                        title="Missing Content-Security-Policy (CSP) Header",
                        severity="medium",
                        rule_id="dast.header.csp",
                        vuln_family="misconfig",
                        cwe="CWE-693",
                        owasp_category="A05:2021-Security Misconfiguration",
                        message=(
                            f"Response from {final_url} does not include a Content-Security-Policy header, "
                            "increasing susceptibility to Cross-Site Scripting (XSS) and data injection."
                        ),
                        file_path=final_url,
                        raw={"missing_header": "Content-Security-Policy", "confidence_level": "high"},
                    )
                )
            elif "'unsafe-inline'" in csp.lower() or "'unsafe-eval'" in csp.lower():
                findings.append(
                    NormalizedFinding(
                        engine="veritas_dast",
                        title="Weak Content-Security-Policy Allows unsafe-inline / unsafe-eval",
                        severity="medium",
                        rule_id="dast.header.csp.weak",
                        vuln_family="xss",
                        cwe="CWE-693",
                        owasp_category="A05:2021-Security Misconfiguration",
                        message=f"CSP header on {final_url} contains permissive directives: {csp[:180]}",
                        file_path=final_url,
                        snippet=f"Content-Security-Policy: {csp[:240]}",
                        raw={"csp": csp[:240], "confidence_level": "high"},
                    )
                )

            # 3. Strict-Transport-Security (HSTS) check
            if final_scheme == "https" and not headers.get("strict-transport-security"):
                findings.append(
                    NormalizedFinding(
                        engine="veritas_dast",
                        title="Missing Strict-Transport-Security (HSTS) Header",
                        severity="medium",
                        rule_id="dast.header.hsts",
                        vuln_family="misconfig",
                        cwe="CWE-319",
                        owasp_category="A05:2021-Security Misconfiguration",
                        message=(
                            f"HTTPS endpoint {final_url} omits Strict-Transport-Security, allowing SSL-stripping "
                            "downgrade attacks on first-time connections."
                        ),
                        file_path=final_url,
                        raw={"missing_header": "Strict-Transport-Security", "confidence_level": "high"},
                    )
                )

            # 4. Anti-Clickjacking check (X-Frame-Options / frame-ancestors)
            xfo = headers.get("x-frame-options")
            has_frame_ancestors = bool(csp and "frame-ancestors" in csp.lower())
            if not xfo and not has_frame_ancestors:
                findings.append(
                    NormalizedFinding(
                        engine="veritas_dast",
                        title="Missing Anti-Clickjacking Header (X-Frame-Options / frame-ancestors)",
                        severity="low",
                        rule_id="dast.header.clickjacking",
                        vuln_family="misconfig",
                        cwe="CWE-1021",
                        owasp_category="A05:2021-Security Misconfiguration",
                        message=f"{final_url} can be embedded in arbitrary third-party iframes.",
                        file_path=final_url,
                        raw={"missing_header": "X-Frame-Options", "confidence_level": "high"},
                    )
                )

            # 5. X-Content-Type-Options check
            xcto = (headers.get("x-content-type-options") or "").strip().lower()
            if xcto != "nosniff":
                findings.append(
                    NormalizedFinding(
                        engine="veritas_dast",
                        title="Missing X-Content-Type-Options: nosniff Header",
                        severity="low",
                        rule_id="dast.header.nosniff",
                        vuln_family="misconfig",
                        cwe="CWE-693",
                        owasp_category="A05:2021-Security Misconfiguration",
                        message=f"Response from {final_url} omits 'X-Content-Type-Options: nosniff'.",
                        file_path=final_url,
                        raw={"missing_header": "X-Content-Type-Options", "confidence_level": "high"},
                    )
                )

            # 6. Server / X-Powered-By Version Disclosure check
            server_hdr = headers.get("server") or ""
            powered_by = headers.get("x-powered-by") or ""
            combined_banner = f"{server_hdr} {powered_by}".strip()
            if VERSION_DISCLOSURE_RE.search(combined_banner):
                findings.append(
                    NormalizedFinding(
                        engine="veritas_dast",
                        title=f"Server Technology Version Disclosure ({combined_banner})",
                        severity="low",
                        rule_id="dast.header.version_disclosure",
                        vuln_family="misconfig",
                        cwe="CWE-200",
                        owasp_category="A05:2021-Security Misconfiguration",
                        message=f"Response headers disclose backend stack version: '{combined_banner}'.",
                        file_path=final_url,
                        snippet=f"Server: {server_hdr}\nX-Powered-By: {powered_by}".strip(),
                        raw={"banner": combined_banner, "confidence_level": "high"},
                    )
                )

            # 7. Set-Cookie Security Flags check
            set_cookies = headers.get_list("set-cookie")
            for cookie_raw in set_cookies:
                cookie_name = cookie_raw.split("=", 1)[0].strip() or "session"
                c_low = cookie_raw.lower()
                missing_flags: list[str] = []
                if "httponly" not in c_low:
                    missing_flags.append("HttpOnly")
                if final_scheme == "https" and "secure" not in c_low:
                    missing_flags.append("Secure")
                if "samesite" not in c_low:
                    missing_flags.append("SameSite")

                if missing_flags:
                    cwe = "CWE-1004" if "HttpOnly" in missing_flags else ("CWE-614" if "Secure" in missing_flags else "CWE-352")
                    findings.append(
                        NormalizedFinding(
                            engine="veritas_dast",
                            title=f"Cookie '{cookie_name}' Missing Security Attributes ({', '.join(missing_flags)})",
                            severity="medium" if ("HttpOnly" in missing_flags or "Secure" in missing_flags) else "low",
                            rule_id="dast.cookie.insecure",
                            vuln_family="auth",
                            cwe=cwe,
                            owasp_category="A07:2021-Identification and Authentication Failures",
                            message=(
                                f"Cookie '{cookie_name}' set by {final_url} is missing {', '.join(missing_flags)} "
                                "attributes, exposing it to XSS theft or cross-site transmission."
                            ),
                            file_path=final_url,
                            snippet=f"Set-Cookie: {cookie_raw[:200]}",
                            raw={"cookie": cookie_name, "missing_flags": missing_flags, "confidence_level": "high"},
                        )
                    )

            # 8. CORS Origin Reflection & Credential Theft Probe
            evil_origin = "https://evil-origin.veritas.invalid"
            cors_resp = client.get(validated_url, headers={"Origin": evil_origin})
            acao = (cors_resp.headers.get("access-control-allow-origin") or "").strip()
            acac = (cors_resp.headers.get("access-control-allow-credentials") or "").strip().lower()
            if (acao == evil_origin or acao == "*") and acac == "true":
                findings.append(
                    NormalizedFinding(
                        engine="veritas_dast",
                        title="Permissive CORS Origin Reflection With Credentials Enabled",
                        severity="critical",
                        rule_id="dast.cors.origin_reflection",
                        vuln_family="authz",
                        cwe="CWE-942",
                        owasp_category="A01:2021-Broken Access Control",
                        message=(
                            f"Target {final_url} reflected arbitrary Origin '{acao}' alongside "
                            "'Access-Control-Allow-Credentials: true', allowing cross-origin account takeover."
                        ),
                        file_path=final_url,
                        snippet=f"Access-Control-Allow-Origin: {acao}\nAccess-Control-Allow-Credentials: {acac}",
                        raw={"acao": acao, "acac": acac, "confidence_level": "high"},
                    )
                )

            # 9. Sensitive File & Unauthenticated API Schema Exposure Probes
            base_root = f"{parsed.scheme}://{parsed.netloc}/"
            probes = [
                (
                    ".env",
                    "dast.exposure.dotenv",
                    "Exposed Environment Configuration File (/.env)",
                    "critical",
                    "secret",
                    "CWE-538",
                    lambda body, ct: (not _looks_like_html_fallback(body, ct)) and bool(ENV_KEY_VALUE_RE.search(body)),
                ),
                (
                    ".git/HEAD",
                    "dast.exposure.git",
                    "Exposed Git Repository Metadata (/.git/HEAD)",
                    "high",
                    "misconfig",
                    "CWE-538",
                    lambda body, ct: (not _looks_like_html_fallback(body, ct)) and ("ref: refs/" in body),
                ),
                (
                    "openapi.json",
                    "dast.exposure.openapi",
                    "Publicly Exposed OpenAPI Specification (/openapi.json)",
                    "low",
                    "misconfig",
                    "CWE-200",
                    lambda body, ct: (not _looks_like_html_fallback(body, ct)) and ('"openapi"' in body or '"swagger"' in body),
                ),
                (
                    "swagger.json",
                    "dast.exposure.swagger",
                    "Publicly Exposed Swagger API Schema (/swagger.json)",
                    "low",
                    "misconfig",
                    "CWE-200",
                    lambda body, ct: (not _looks_like_html_fallback(body, ct)) and ('"swagger"' in body or '"openapi"' in body),
                ),
            ]

            for rel_path, rule_id, title, sev, fam, cwe, validator in probes:
                probe_url = urljoin(base_root, rel_path)
                try:
                    p_resp = client.get(probe_url)
                    if p_resp.status_code == 200:
                        body_text = p_resp.text[:4000]
                        ct = p_resp.headers.get("content-type") or ""
                        if validator(body_text, ct):
                            findings.append(
                                NormalizedFinding(
                                    engine="veritas_dast",
                                    title=title,
                                    severity=sev,
                                    rule_id=rule_id,
                                    vuln_family=fam,
                                    cwe=cwe,
                                    owasp_category="A05:2021-Security Misconfiguration",
                                    message=f"Sensitive endpoint {probe_url} returned HTTP 200 with verified signature.",
                                    file_path=probe_url,
                                    snippet=body_text[:240],
                                    raw={"probe_url": probe_url, "confidence_level": "high"},
                                )
                            )
                except Exception:
                    continue

    except Exception as exc:
        logger.warning("DAST probe encountered network error for %s: %s", validated_url, str(exc)[:200])
        meta["error"] = str(exc)[:200]

    meta["findings"] = len(findings)
    return findings, meta

def _build_deterministic_diff(finding: Finding) -> str:
    file_path = (finding.file_path or "src/app.py").lstrip("./")
    raw_data: dict[str, Any] = {}
    if finding.raw_json:
        try:
            raw_data = json.loads(finding.raw_json)
        except Exception:
            raw_data = {}

    pkg = (
        raw_data.get("package_name")
        or raw_data.get("PkgName")
        or raw_data.get("package")
    )
    installed_ver = (
        raw_data.get("installed_version")
        or raw_data.get("InstalledVersion")
        or raw_data.get("version")
    )
    fixed_ver = (
        raw_data.get("fixed_version")
        or raw_data.get("FixedVersion")
        or (raw_data.get("fix_versions")[0] if isinstance(raw_data.get("fix_versions"), list) and raw_data.get("fix_versions") else None)
    )

    if finding.vuln_family in ("sca", "dependency", "license") or pkg:
        pkg_name = str(pkg or "package")
        old_ver = str(installed_ver or "1.0.0")
        new_ver = str(fixed_ver or "latest-patched")
        target_file = file_path if not file_path.startswith("http") else "requirements.txt"
        if target_file.endswith("package.json"):
            return (
                f"--- a/{target_file}\n"
                f"+++ b/{target_file}\n"
                f"@@ -1,3 +1,3 @@\n"
                f'-    "{pkg_name}": "{old_ver}"\n'
                f'+    "{pkg_name}": "^{new_ver}"\n'
            )
        return (
            f"--- a/{target_file}\n"
            f"+++ b/{target_file}\n"
            f"@@ -1,1 +1,1 @@\n"
            f"-{pkg_name}=={old_ver}\n"
            f"+{pkg_name}>={new_ver}\n"
        )

    if finding.vuln_family in ("secret", "secrets"):
        line_no = finding.line_start or 1
        snippet = (finding.snippet or "API_KEY = 'REDACTED'").splitlines()[0]
        return (
            f"--- a/{file_path}\n"
            f"+++ b/{file_path}\n"
            f"@@ -{line_no},1 +{line_no},2 @@\n"
            f"-{snippet}\n"
            f"+import os\n"
            f"+# Loaded securely from environment secret manager\n"
        )

"""Persist + query HTTP request telemetry and Active IPS Auto-Quarantine."""

from __future__ import annotations

import logging
import threading
import time
from collections import Counter
from datetime import timedelta

from sqlalchemy import delete, func, select

from app.core import database as db

logger = logging.getLogger(__name__)

MAX_EVENTS_RETENTION = 50_000
RETENTION_DAYS = 7
_WRITE_COUNTER = 0

IPS_WINDOW_SECONDS = 60
IPS_ATTACK_THRESHOLD = 5
IPS_QUARANTINE_MINUTES = 60
ATTACK_CATEGORIES = {"sqli", "xss", "ssrf", "cmd_injection", "cmdi", "path_traversal"}

_IPS_LOCK = threading.Lock()
_BLOCKED_IP_CACHE: dict[str, tuple[float | None, str]] = {}
_CACHE_LAST_SYNC: float = 0.0


def refresh_blocked_ip_cache(session: Session | None = None) -> None:
    """Syncs active `BlockedIp` rows into the in-memory lookup table."""
    global _CACHE_LAST_SYNC
    own_session = session is None
    sess = session or db.SessionLocal()
    try:
        now = utcnow()
        rows = list(sess.scalars(select(BlockedIp)).all())
        new_map: dict[str, tuple[float | None, str]] = {}
        for r in rows:
            exp = ensure_utc(r.expires_at) if r.expires_at else None
            if exp and exp < now:
                continue
            exp_ts = exp.timestamp() if exp else None
            new_map[r.ip_address] = (exp_ts, r.reason)
        with _IPS_LOCK:
            _BLOCKED_IP_CACHE.clear()
            _BLOCKED_IP_CACHE.update(new_map)
            _CACHE_LAST_SYNC = time.monotonic()
    except Exception as exc:
        logger.debug("BlockedIp cache refresh skipped: %s", exc)
    finally:
        if own_session:
            sess.close()


def is_ip_quarantined(ip: str | None) -> tuple[bool, str | None]:
    if not ip:
        return False, None
    now_mono = time.monotonic()
    if now_mono - _CACHE_LAST_SYNC > 5.0:
        refresh_blocked_ip_cache()

    now_epoch = utcnow().timestamp()
    with _IPS_LOCK:
        entry = _BLOCKED_IP_CACHE.get(ip)
        if not entry:
            return False, None
        exp_ts, reason = entry
        if exp_ts is not None and now_epoch > exp_ts:
            _BLOCKED_IP_CACHE.pop(ip, None)
            return False, None
        return True, reason


def block_ip_address(
    session: Session,
    *,
    ip_address: str,
    reason: str,
    attack_category: str = "manual",
    hit_count: int = 1,
    expires_in_minutes: int | None = IPS_QUARANTINE_MINUTES,
    created_by: str = "admin",
) -> BlockedIp:
    clean_ip = ip_address.strip()[:64]
    now = utcnow()
    expires_at = (
        now + timedelta(minutes=int(expires_in_minutes))
        if expires_in_minutes is not None and int(expires_in_minutes) > 0
        else None
    )
    row = session.scalar(select(BlockedIp).where(BlockedIp.ip_address == clean_ip))
    if row is None:
        row = BlockedIp(
            ip_address=clean_ip,
            reason=reason[:512],
            attack_category=attack_category[:64],
            hit_count=hit_count,
            blocked_at=now,
            expires_at=expires_at,
            created_by=created_by[:64],
        )
        session.add(row)
    else:
        row.reason = reason[:512]
        row.attack_category = attack_category[:64]
        row.hit_count = max(row.hit_count, hit_count)
        row.blocked_at = now
        row.expires_at = expires_at
        row.created_by = created_by[:64]
        session.add(row)

    session.commit()
    session.refresh(row)
    refresh_blocked_ip_cache(session)
    return row


def unblock_ip_address(session: Session, ip_address: str) -> bool:
    clean_ip = ip_address.strip()[:64]
    row = session.scalar(select(BlockedIp).where(BlockedIp.ip_address == clean_ip))
    if not row:
        with _IPS_LOCK:
            _BLOCKED_IP_CACHE.pop(clean_ip, None)
        return False
    session.delete(row)
    session.commit()
    with _IPS_LOCK:
        _BLOCKED_IP_CACHE.pop(clean_ip, None)
    return True

def record_http_event(
    *,
    method: str,
    path: str,
    query: str | None,
    status_code: int,
    duration_ms: float,
    ip: str | None,
    user_agent: str | None,
    user_id: str | None = None,
) -> None:
    """Best-effort writer + Active IPS auto-quarantine detector — never raises into request path."""
    global _WRITE_COUNTER
    classification = classify_request(
        method=method,
        path=path,
        query=query,
        user_agent=user_agent,
        status_code=status_code,
    )
    try:
        with db.SessionLocal() as session:
            now = utcnow()
            session.add(
                HttpRequestEvent(
                    method=method[:12],
                    path=path[:512],
                    query=(query[:512] if query else None),
                    status_code=int(status_code),

                    ip=(ip[:64] if ip else None),
                    user_agent=(user_agent[:256] if user_agent else None),
                    user_id=user_id,
                    attack_category=classification.category,
                    severity=classification.severity,
                    matched_pattern=classification.matched_pattern,
                )
            )
            session.commit()

            # Phase 6C Active IPS: if IP triggered >= 5 attack events within 60 seconds, auto-quarantine
            if ip and classification.category in ATTACK_CATEGORIES:
                window_start = now - timedelta(seconds=IPS_WINDOW_SECONDS)
                recent_hits = (
                    session.scalar(
                        select(func.count(HttpRequestEvent.id)).where(
                            HttpRequestEvent.ip == ip[:64],
                            HttpRequestEvent.attack_category.in_(sorted(ATTACK_CATEGORIES)),
                            HttpRequestEvent.created_at >= window_start,
                        )
                    )
                    or 0
                )
                if int(recent_hits) >= IPS_ATTACK_THRESHOLD:
                    block_ip_address(
                        session,
                        ip_address=ip[:64],
                        reason=(
                            f"Auto-quarantined by VERITAS Active IPS after {recent_hits} "
                            f"attack payloads (latest: {classification.category})"
                        ),
                        attack_category=classification.category,
                        hit_count=int(recent_hits),
                        expires_in_minutes=IPS_QUARANTINE_MINUTES,
                        created_by="auto_ips",
                    )

            _WRITE_COUNTER += 1
            if _WRITE_COUNTER % 200 == 0:
                _prune_old_events(session)
    except Exception as exc:  # pragma: no cover - defensive
        logger.debug("http telemetry write skipped: %s", exc)


def _prune_old_events(session: Session) -> None:
    cutoff = utcnow() - timedelta(days=RETENTION_DAYS)
    session.execute(delete(HttpRequestEvent).where(HttpRequestEvent.created_at < cutoff))
    total = session.scalar(select(func.count(HttpRequestEvent.id))) or 0
    if total > MAX_EVENTS_RETENTION:
        sub = (
            select(HttpRequestEvent.id)
            .order_by(HttpRequestEvent.created_at.desc())
            .offset(MAX_EVENTS_RETENTION)
        )
        session.execute(delete(HttpRequestEvent).where(HttpRequestEvent.id.in_(sub)))
    session.commit()

def http_traffic_stats(session: Session, *, since_minutes: int = 60) -> dict:
    window = max(5, min(since_minutes, 60 * 24 * 7))
    now = utcnow()
    cutoff = now - timedelta(minutes=window)
    rows = list(
        session.scalars(
            select(HttpRequestEvent)
            .where(HttpRequestEvent.created_at >= cutoff)
            .order_by(HttpRequestEvent.created_at.asc())
        )
    )

    bucket_minutes = 1 if window <= 60 else (5 if window <= 360 else 30)
    bucket_seconds = bucket_minutes * 60
    start_epoch = int(cutoff.timestamp() // bucket_seconds * bucket_seconds)
    end_epoch = int(now.timestamp() // bucket_seconds * bucket_seconds)

    buckets: dict[int, dict[str, int]] = {}
    t = start_epoch
    while t <= end_epoch:
        buckets[t] = {"total": 0, "suspicious": 0, "errors": 0}
        t += bucket_seconds

    by_category: Counter[str] = Counter()
    by_severity: Counter[str] = Counter()
    by_status: Counter[str] = Counter()
    ip_total: Counter[str] = Counter()
    ip_suspicious: Counter[str] = Counter()
    path_counts: Counter[str] = Counter()
    total_duration = 0.0
    suspicious_total = 0
    error_total = 0

    for r in rows:
        ts = int(ensure_utc(r.created_at).timestamp() // bucket_seconds * bucket_seconds)
        b = buckets.setdefault(ts, {"total": 0, "suspicious": 0, "errors": 0})
        b["total"] += 1
        is_susp = r.attack_category != "normal"
        if is_susp:
            b["suspicious"] += 1
            suspicious_total += 1
        if r.status_code >= 400:
            b["errors"] += 1
            error_total += 1
        by_category[r.attack_category] += 1
        by_severity[r.severity] += 1
        by_status[f"{r.status_code // 100}xx"] += 1
        if r.ip:
            ip_total[r.ip] += 1
            if is_susp:
                ip_suspicious[r.ip] += 1
        path_counts[r.path] += 1
        total_duration += float(r.duration_ms or 0.0)

    timeline = [
        {"ts": ts * 1000, "total": vals["total"], "suspicious": vals["suspicious"], "errors": vals["errors"]}
        for ts, vals in sorted(buckets.items())
    ]

    top_ips = [
        {"ip": ip, "requests": count, "suspicious": ip_suspicious.get(ip, 0)}
        for ip, count in ip_total.most_common(8)
    ]
    top_suspicious_ips = [
        {"ip": ip, "suspicious": count, "total": ip_total.get(ip, count)}
        for ip, count in ip_suspicious.most_common(8)
    ]
    top_paths = [{"path": p, "requests": c} for p, c in path_counts.most_common(8)]

    total = len(rows)
    avg_ms = round(total_duration / total, 2) if total else 0.0

    return {
        "window_minutes": window,
        "bucket_minutes": bucket_minutes,
        "totals": {
            "requests": total,
            "suspicious": suspicious_total,
            "errors": error_total,
            "unique_ips": len(ip_total),
            "avg_duration_ms": avg_ms,
        },
        "timeline": timeline,
        "by_category": [
            {"category": k, "label": ATTACK_LABELS.get(k, k), "count": v}
            for k, v in by_category.most_common()
        ],
        "by_severity": dict(by_severity),
        "by_status": dict(by_status),
        "top_ips": top_ips,
        "top_suspicious_ips": top_suspicious_ips,
        "top_paths": top_paths,
    }

"""Starlette middleware capturing every HTTP request for the security monitor and enforcing Active IPS."""

from __future__ import annotations

import time
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import JSONResponse, Response

from app.services.http_traffic import is_ip_quarantined, record_http_event

_SKIP_PREFIXES = (
    "/assets/",
    "/static/",
    "/brand/",
    "/admin/http-events",
    "/admin/security/blocked-ips",
)
_SKIP_EXACT = {
    "/favicon.ico",
    "/favicon.png",
    "/favicon.jpg",
    "/apple-touch-icon.png",
    "/apple-touch-icon.jpg",
    "/site.webmanifest",
    "/robots.txt",
    "/sitemap.xml",
    "/icons.svg",
}


def _client_ip(request: Request) -> str | None:
    xff = request.headers.get("x-forwarded-for")
    if xff:
        first = xff.split(",")[0].strip()
        if first:
            return first
    real_ip = request.headers.get("x-real-ip")
    if real_ip:
        return real_ip.strip()
    if request.client and request.client.host:
        return request.client.host
    return None


class HttpTelemetryMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next) -> Response:
        path = request.url.path or "/"
        if path in _SKIP_EXACT or any(path.startswith(p) for p in _SKIP_PREFIXES):
            return await call_next(request)

        ip = _client_ip(request)
        # Enforce Active IPS Quarantine (never block /admin/* or /auth/* so admins can log in and unblock)
        if ip and not path.startswith("/admin") and not path.startswith("/auth"):
            blocked, reason = is_ip_quarantined(ip)
            if blocked:
                return JSONResponse(
                    status_code=403,
                    content={
                        "detail": "IP address temporarily quarantined by VERITAS Active IPS",
                        "reason": reason,
                    },
                )

        start = time.perf_counter()
        status_code = 500
        try:
            response = await call_next(request)
            status_code = response.status_code
            return response
        finally:
            duration_ms = (time.perf_counter() - start) * 1000.0
            record_http_event(
                method=request.method,
                path=path,
                query=request.url.query or None,
                status_code=status_code,
                duration_ms=duration_ms,
                ip=ip,
                user_agent=request.headers.get("user-agent"),
                user_id=getattr(request.state, "user_id", None),
            )

import React, { useMemo } from "react";
import type { Finding } from "../../lib/workspace";

export type RiskCellCoord = { likelihood: number; impact: number };

export function computeFindingRiskCell(finding: Finding): RiskCellCoord {
  const sev = (finding.severity || "info").toLowerCase();
  const impact =
    sev === "critical" ? 5 : sev === "high" ? 4 : sev === "medium" ? 3 : sev === "low" ? 2 : 1;

  const epss = finding.epss_score ?? 0;
  const conf = (finding.confidence_level || "medium").toLowerCase();
  const engineCount = finding.engine_count ?? finding.matched_engines?.length ?? 1;

  let likelihood = 2;
  if (finding.is_kev || epss >= 0.3) {
    likelihood = 5;
  } else if (epss >= 0.1 || (conf === "high" && engineCount >= 2) || finding.engine === "veritas_dast") {
    likelihood = 4;
  } else if (epss >= 0.02 || conf === "high") {
    likelihood = 3;
  } else if (conf === "medium") {
    likelihood = 2;
  } else {
    likelihood = 1;
  }

  return { likelihood, impact };
}

function cellZoneStyle(likelihood: number, impact: number, isSelected: boolean): React.CSSProperties {
  const product = likelihood * impact;
  let bg = "rgba(16, 185, 129, 0.14)";
  let border = "rgba(52, 211, 153, 0.35)";
  let color = "#6ee7b7";

  if (product >= 16 || (impact === 5 && likelihood >= 3)) {
    bg = "rgba(220, 38, 38, 0.28)";
    border = "rgba(248, 113, 113, 0.65)";
    color = "#fecaca";
  } else if (product >= 10) {
    bg = "rgba(249, 115, 22, 0.24)";
    border = "rgba(251, 146, 60, 0.55)";
    color = "#fed7aa";
  } else if (product >= 5) {
    bg = "rgba(245, 158, 11, 0.18)";
    border = "rgba(251, 191, 36, 0.45)";
    color = "#fde68a";
  }

  return {
    backgroundColor: bg,
    border: isSelected ? "2px solid #ffffff" : `1px solid ${border}`,
    color,
    borderRadius: 8,
    padding: "8px 4px",
    textAlign: "center",
    cursor: "pointer",
    fontWeight: 700,
    fontSize: "0.84rem",
    transition: "transform 0.12s ease",
  };
}

  const { grid, extremeCount, highCount, meanScore, residualScore } = useMemo(() => {
    const matrix: Record<string, number> = {};
    let extreme = 0;
    let high = 0;
    let totalScore = 0;
    let openScore = 0;
    let openCount = 0;

    for (const f of findings) {
      const score = f.effective_priority_score ?? f.risk_score ?? 0;
      totalScore += score;
      if (f.status !== "false_positive" && f.status !== "fixed") {
        openScore += score;
        openCount += 1;
        const { likelihood, impact } = computeFindingRiskCell(f);
        const key = `${likelihood}-${impact}`;
        matrix[key] = (matrix[key] || 0) + 1;
        const prod = likelihood * impact;
        if (prod >= 16 || (impact === 5 && likelihood >= 3)) extreme += 1;
        else if (prod >= 10) high += 1;
      }
    }

    return {
      grid: matrix,
      extremeCount: extreme,
      highCount: high,
      meanScore: findings.length ? (totalScore / findings.length).toFixed(1) : "0.0",
      residualScore: openCount ? (openScore / openCount).toFixed(1) : "0.0",
    };
  }, [findings]);

  return (

        marginBottom: 16,

        <div>
          <h4 style={{ margin: 0, fontSize: "0.95rem" }}>5×5 Quantitative Risk Assessment Matrix (Likelihood × Impact)</h4>
          <span className="muted small">Click any cell to filter findings by Likelihood & Impact tier</span>
        </div>
        <div style={{ display: "flex", gap: 8, flexWrap: "wrap", alignItems: "center" }}>
          <span className="badge sev-critical">Extreme: {extremeCount}</span>
          <span className="badge sev-high">High: {highCount}</span>
          <span className="badge sev-info">Mean Priority: {meanScore}/10</span>
          <span className="badge sev-info">Residual Risk: {residualScore}/10</span>
          {selectedCell ? (
            <button type="button" className="btn btn-ghost btn-xs" onClick={() => onSelectCell(null)}>

            </button>

              </div>
